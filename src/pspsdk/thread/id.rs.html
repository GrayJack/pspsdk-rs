3/// A unique identifier for a running thread.
4///
5/// A `ThreadId` is an opaque object that uniquely identifies each thread
6/// created during the lifetime of a process. `ThreadId`s are guaranteed not to
7/// be reused, even when a thread terminates. `ThreadId`s are under the control
8/// of Rust's standard library and there may not be any relationship between
9/// `ThreadId` and the underlying platform's notion of a thread identifier --
10/// the two concepts cannot, therefore, be used interchangeably. A `ThreadId`
11/// can be retrieved from the [`id`] method on a [`Thread`].
12///
13/// # Examples
14///
15/// ```
16/// use pspsdk::thread;
17///
18/// let other_thread = thread::spawn(|| thread::current().id());
19///
20/// let other_thread_id = other_thread.join().unwrap();
21/// assert!(thread::current().id() != other_thread_id);
22/// ```
23///
24/// [`Thread`]: super::Thread
25/// [`id`]: super::Thread::id
26#[derive(Eq, PartialEq, Clone, Copy, Hash, Debug)]
27pub struct ThreadId(NonZero<u64>);
28
29impl ThreadId {
30    // Generate a new unique thread ID.
31    pub(crate) fn new() -> ThreadId {
32        #[cold]
33        fn exhausted() -> ! {
34            panic!("failed to generate unique thread ID: bitspace exhausted")
35        }
36
37        // Here we will diverge from std and take advantage we are on a PSP and update the 64bit ID
38        // during disabled interrupts. This removes the need to do spin-locks even though the system
39        // doesn't have 64bit atomics. If you want to know how Rust STD does, look up for the
40        // similar file on Rust source.
41
42        static COUNTER: SyncUnsafeCell<u64> = SyncUnsafeCell::new(0);
43
44
45        let id = crate::process::with_suspended_interrupts(|| {
46            let last = unsafe { COUNTER.get().read_volatile() };
47
48            let id = last.checked_add(1)?;
49
50            unsafe { COUNTER.get().write(id) };
51            Some(ThreadId(unsafe { NonZero::new_unchecked(id) }))
52        });
53
54        match id {
55            Some(id) => id,
56            None => exhausted(),
57        }
58    }
59
60    // #[cfg(any(not(target_thread_local), target_has_atomic = "64"))]
61    pub(super) fn from_u64(v: u64) -> Option<ThreadId> {
62        NonZero::new(v).map(ThreadId)
63    }
64
65    /// This returns a numeric identifier for the thread identified by this
66    /// `ThreadId`.
67    ///
68    /// As noted in the documentation for the type itself, it is essentially an
69    /// opaque ID, but is guaranteed to be unique for each thread. The returned
70    /// value is entirely opaque -- only equality testing is stable. Note that
71    /// it is not guaranteed which values new threads will return, and this may
72    /// change across Rust versions.
73    #[must_use]
74    pub fn as_u64(&self) -> NonZero<u64> {
75        self.0
76    }
77}
