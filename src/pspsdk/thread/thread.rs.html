46/// The internal representation of a `Thread` handle
47///
48/// We explicitly set the alignment for our guarantee in Thread::into_raw. This
49/// allows applications to stuff extra metadata bits into the alignment, which
50/// can be rather useful when working with atomics.
51#[repr(align(8))]
52struct Inner {
53    name: Option<ThreadNameString>,
54    id: ThreadId,
55    os_id: OnceLock<u64>,
56    parker: Parker,
57}
58
59impl Inner {
60    fn parker(self: Pin<&Self>) -> Pin<&Parker> {
61        unsafe { Pin::map_unchecked(self, |inner| &inner.parker) }
62    }
63}

65/// A handle to a thread.
66///
67/// Threads are represented via the `Thread` type, which you can get in one of
68/// two ways:
69///
70/// * By spawning a new thread, e.g., using the [`thread::spawn`] function, and calling [`thread`]
71///   on the [`JoinHandle`].
72/// * By requesting the current thread, using the [`thread::current`] function.
73///
74/// The [`thread::current`] function is available even for threads not spawned
75/// by the APIs of this module.
76///
77/// There is usually no need to create a `Thread` struct yourself, one
78/// should instead use a function like `spawn` to create new threads, see the
79/// docs of [`Builder`] and [`spawn`] for more details.
80///
81/// [`thread::spawn`]: super::spawn
82/// [`thread`]: super::JoinHandle::thread
83/// [`JoinHandle`]: super::JoinHandle
84/// [`thread::current`]: super::current::current
85/// [`Builder`]: super::Builder
86/// [`spawn`]: super::spawn
87#[derive(Clone)]
88pub struct Thread {
89    // We use the System allocator such that creating or dropping this handle
90    // does not interfere with a potential Global allocator using thread-local
91    // storage.
92    inner: Pin<Arc<Inner, System>>,
93}
94
95impl Thread {
96    pub(crate) fn new(id: ThreadId, name: Option<String>) -> Thread {
97        let name = name.map(ThreadNameString::from);
98
99        // We have to use `unsafe` here to construct the `Parker` in-place,
100        // which is required for the UNIX implementation.
101        //
102        // SAFETY: We pin the Arc immediately after creation, so its address never
103        // changes.
104        let inner = unsafe {
105            let mut arc = Arc::<Inner, _>::new_uninit_in(System);
106            let ptr = Arc::get_mut_unchecked(&mut arc).as_mut_ptr();
107            (&raw mut (*ptr).name).write(name);
108            (&raw mut (*ptr).id).write(id);
109            (&raw mut (*ptr).os_id).write(OnceLock::new());
110            Parker::new_in_place(&raw mut (*ptr).parker);
111            Pin::new_unchecked(arc.assume_init())
112        };
113
114        Thread { inner }
115    }

117    /// Creates a handle for the calling thread, recording its OS id.
118    ///
119    /// `id` must be the `ThreadId` of the calling thread.
120    ///
121    /// Takes no name because passing one into `Thread::new` allocates with the
122    /// global allocator, which `thread::current` is documented never to use.
123    pub(crate) fn new_current(id: ThreadId) -> Thread {
124        let thread = Thread::new(id, None);
125        thread.set_os_id_to_current();
126        thread
127    }

129    /// Records the calling thread's OS id, as reported by
130    /// `imp::current_os_id`, in this handle.
131    ///
132    /// May only be called from the thread to which this handle belongs. A
133    /// spawned thread does this itself once it starts running, since its handle
134    /// already exists by then.
135    ///
136    /// `imp::current_os_id` must not allocate with the global allocator or call
137    /// `thread::current`.
138    pub(crate) fn set_os_id_to_current(&self) {
139        if let Some(os_id) = imp::current_os_id()
140            && self.inner.os_id.set(os_id).is_err()
141        {
142            crate::rtabort!("thread OS id already set");
143        }
144    }

146    /// Like the public [`park`], but callable on any handle. This is used to
147    /// allow parking in TLS destructors.
148    ///
149    /// # Safety
150    /// May only be called from the thread to which this handle belongs.
151    ///
152    /// [`park`]: super::park
153    pub(crate) unsafe fn park(&self) {
154        unsafe { self.inner.as_ref().parker().park() }

162    ///
163    /// [`park_timeout`]: super::park_timeout

165        unsafe { self.inner.as_ref().parker().park_timeout(dur) }
166    }

168    /// Atomically makes the handle's token available if it is not already.
169    ///
170    /// Every thread is equipped with some basic low-level blocking support, via
171    /// the [`park`] function and the `unpark()` method. These can be used as a
172    /// more CPU-efficient implementation of a spinlock.
173    ///
174    /// See the [park documentation] for more details.
175    ///
176    /// # Examples
177    ///
178    /// ```
179    /// use std::{
180    ///     sync::atomic::{AtomicBool, Ordering},
181    ///     thread,
182    ///     time::Duration,
183    /// };
184    ///
185    /// static QUEUED: AtomicBool = AtomicBool::new(false);
186    ///
187    /// let parked_thread = thread::Builder::new()
188    ///     .spawn(|| {
189    ///         println!("Parking thread");
190    ///         QUEUED.store(true, Ordering::Release);
191    ///         thread::park();
192    ///         println!("Thread unparked");
193    ///     })
194    ///     .unwrap();
195    ///
196    /// // Let some time pass for the thread to be spawned.
197    /// thread::sleep(Duration::from_millis(10));
198    ///
199    /// // Wait until the other thread is queued.
200    /// // This is crucial! It guarantees that the `unpark` below is not consumed
201    /// // by some other code in the parked thread (e.g. inside `println!`).
202    /// while !QUEUED.load(Ordering::Acquire) {
203    ///     // Spinning is of course inefficient; in practice, this would more likely be
204    ///     // a dequeue where we have no work to do if there's nobody queued.
205    ///     std::hint::spin_loop();
206    /// }
207    ///
208    /// println!("Unpark the thread");
209    /// parked_thread.thread().unpark();
210    ///
211    /// parked_thread.join().unwrap();
212    /// ```
213    ///
214    /// [`park`]: super::park
215    /// [park documentation]: super::park
216    #[inline]
217    pub fn unpark(&self) {
218        self.inner.as_ref().parker().unpark();
219    }

238    /// Gets the id the operating system gave this thread, if it has one that can
239    /// be read.
240    ///
241    /// This is the id that shows up in tools like `ps` and `top`, debuggers and
242    /// crash logs, unlike [`ThreadId`], which has no guaranteed relationship to
243    /// it. On a platform with no OS-visible thread id, such as SGX, the value
244    /// may be some other per-thread value (there, the thread's address), which
245    /// such tools will not recognize. `None` means no id could be recorded: the
246    /// thread has not started running yet, or the platform has no way to read
247    /// one.
248    ///
249    /// The operating system may reuse the id of a thread that has exited, and a
250    /// `Thread` handle can outlive the thread it refers to. After a `fork`, the
251    /// id recorded in the child process still refers to the parent's thread; it
252    /// is not re-read. Use the id only where a reused or stale id is harmless,
253    /// such as logging.
254    ///
255    /// # Examples
256    ///
257    /// ```
258    /// use pspsdk::thread;
259    ///
260    /// let spawned = thread::spawn(|| thread::current().os_id()).join().unwrap();
261    /// if spawned.is_some() {
262    ///     assert_ne!(spawned, thread::current().os_id());
263    /// }
264    /// ```
265    #[must_use]
266    pub fn os_id(&self) -> Option<u64> {
267        self.inner.os_id.get().copied()
268    }

270    /// Gets the thread's name.
271    ///
272    /// For more information about named threads, see
273    /// [this module-level documentation][naming-threads].
274    ///
275    /// # Examples
276    ///
277    /// Threads by default have no name specified:
278    ///
279    /// ```
280    /// use pspsdk::thread;
281    ///
282    /// let builder = thread::Builder::new();
283    ///
284    /// let handler = builder
285    ///     .spawn(|| {
286    ///         assert!(thread::current().name().is_none());
287    ///     })
288    ///     .unwrap();
289    ///
290    /// handler.join().unwrap();
291    /// ```
292    ///
293    /// Thread with a specified name:
294    ///
295    /// ```
296    /// use pspsdk::thread;
297    ///
298    /// let builder = thread::Builder::new().name("foo".into());
299    ///
300    /// let handler = builder
301    ///     .spawn(|| assert_eq!(thread::current().name(), Some("foo")))
302    ///     .unwrap();
303    ///
304    /// handler.join().unwrap();
305    /// ```
306    ///
307    /// [naming-threads]: ./index.html#naming-threads
308    #[must_use]
309    pub fn name(&self) -> Option<&str> {
310        if let Some(name) = &self.inner.name {
311            Some(name.as_str())
312        } else if main_thread::get() == Some(self.inner.id) {
313            Some("main")
314        } else {
315            None
316        }
317    }

319    /// Consumes the `Thread`, returning a raw pointer.
320    ///
321    /// To avoid a memory leak the pointer must be converted
322    /// back into a `Thread` using [`Thread::from_raw`]. The pointer is
323    /// guaranteed to be aligned to at least 8 bytes.
324    ///
325    /// # Examples
326    ///
327    /// ```
328    /// use pspsdk::thread::{self, Thread};
329    ///
330    /// let thread = thread::current();
331    /// let id = thread.id();
332    /// let ptr = Thread::into_raw(thread);
333    /// unsafe {
334    ///     assert_eq!(Thread::from_raw(ptr).id(), id);
335    /// }
336    /// ```
337    pub fn into_raw(self) -> *const () {
338        // Safety: We only expose an opaque pointer, which maintains the `Pin` invariant.
339        let inner = unsafe { Pin::into_inner_unchecked(self.inner) };
340        Arc::into_raw_with_allocator(inner).0 as *const ()
341    }
342
343    /// Constructs a `Thread` from a raw pointer.
344    ///
345    /// The raw pointer must have been previously returned
346    /// by a call to [`Thread::into_raw`].
347    ///
348    /// # Safety
349    ///
350    /// This function is unsafe because improper use may lead
351    /// to memory unsafety, even if the returned `Thread` is never
352    /// accessed.
353    ///
354    /// Creating a `Thread` from a pointer other than one returned
355    /// from [`Thread::into_raw`] is **undefined behavior**.
356    ///
357    /// Calling this function twice on the same raw pointer can lead
358    /// to a double-free if both `Thread` instances are dropped.
359    pub unsafe fn from_raw(ptr: *const ()) -> Thread {
360        // Safety: Upheld by caller.
361        unsafe {
362            Thread {
363                inner: Pin::new_unchecked(Arc::from_raw_in(ptr as *const Inner, System)),
364            }
365        }
366    }
