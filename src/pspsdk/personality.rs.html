1//! Implementation of panics backed by libgcc/libunwind (in some form).
2//!
3//! For background on exception handling and stack unwinding please see
4//! "Exception Handling in LLVM" (llvm.org/docs/ExceptionHandling.html) and
5//! documents linked from it.
6//! These are also good reads:
7//!  * <https://itanium-cxx-abi.github.io/cxx-abi/abi-eh.html>
8//!  * <https://nicolasbrailo.github.io/blog/projects_texts/13exceptionsunderthehood.html>
9//!  * <https://www.airs.com/blog/index.php?s=exception+frames>
10//!
11//! ## A brief summary
12//!
13//! Exception handling happens in two phases: a search phase and a cleanup
14//! phase.
15//!
16//! In both phases the unwinder walks stack frames from top to bottom using
17//! information from the stack frame unwind sections of the current process's
18//! modules ("module" here refers to an OS module, i.e., an executable or a
19//! dynamic library).
20//!
21//! For each stack frame, it invokes the associated "personality routine", whose
22//! address is also stored in the unwind info section.
23//!
24//! In the search phase, the job of a personality routine is to examine
25//! exception object being thrown, and to decide whether it should be caught at
26//! that stack frame. Once the handler frame has been identified, cleanup phase
27//! begins.
28//!
29//! In the cleanup phase, the unwinder invokes each personality routine again.
30//! This time it decides which (if any) cleanup code needs to be run for
31//! the current stack frame. If so, the control is transferred to a special
32//! branch in the function body, the "landing pad", which invokes destructors,
33//! frees memory, etc. At the end of the landing pad, control is transferred
34//! back to the unwinder and unwinding resumes.
35//!
36//! Once stack has been unwound down to the handler frame level, unwinding stops
37//! and the last personality routine transfers control to the catch block.

104#[inline(always)]
105#[cfg(all(not(feature = "std"), panic = "unwind"))]
106unsafe fn sign_lpad(context: *mut uw::_Unwind_Context, lpad: *const u8) -> *const u8 {
107    cfg_select! {
108        all(target_abi = "pauthtest", target_arch = "aarch64") => {
109            // DWARF register number for SP on AArch64.
110            const SP_REG: i32 = 31;
111
112            unsafe {
113                let sp = uw::_Unwind_GetGR(context, SP_REG).addr() as u64;
114                let mut addr = lpad.addr();
115
116                // `pacib` corresponds to `ptrauth_key_process_dependent_code` in
117                // <ptrauth.h>.
118                core::arch::asm!(
119                    "pacib {addr}, {sp}",
120                    addr = inout(reg) addr,
121                    sp = in(reg) sp,
122                    options(nostack, preserves_flags)
123                );
124
125                lpad.with_addr(addr)
126            }
127        },
128        _ => {
129            let _ = context;
130            lpad
131        },
132    }
133}

135/// Default personality routine, which is used directly on most targets
136/// and indirectly on Windows x86_64 and AArch64 via SEH.
137#[cfg(all(not(feature = "std"), panic = "unwind"))]
138unsafe extern "C" fn rust_eh_personality_impl(
139    version: c_int, actions: uw::_Unwind_Action, _exception_class: uw::_Unwind_Exception_Class,
140    exception_object: *mut uw::_Unwind_Exception, context: *mut uw::_Unwind_Context,
141) -> uw::_Unwind_Reason_Code {
142    unsafe {
143        if version != 1 {
144            return uw::_URC_FATAL_PHASE1_ERROR;
145        }
146        let eh_action = match find_eh_action(context) {
147            Ok(action) => action,
148            Err(_) => return uw::_URC_FATAL_PHASE1_ERROR,
149        };
150        if actions & uw::_UA_SEARCH_PHASE != 0 {
151            match eh_action {
152                EHAction::None | EHAction::Cleanup(_) => uw::_URC_CONTINUE_UNWIND,
153                EHAction::Catch(_) | EHAction::Filter(_) => uw::_URC_HANDLER_FOUND,
154                EHAction::Terminate => uw::_URC_FATAL_PHASE1_ERROR,
155            }
156        } else {
157            match eh_action {
158                EHAction::None => uw::_URC_CONTINUE_UNWIND,
159                // Forced unwinding hits a terminate action.
160                EHAction::Filter(_) if actions & uw::_UA_FORCE_UNWIND != 0 => {
161                    uw::_URC_CONTINUE_UNWIND
162                },
163                EHAction::Cleanup(lpad) | EHAction::Catch(lpad) | EHAction::Filter(lpad) => {
164                    uw::_Unwind_SetGR(context, UNWIND_DATA_REG.0, exception_object.cast());
165                    uw::_Unwind_SetGR(context, UNWIND_DATA_REG.1, core::ptr::null());
166                    let maybe_signed_lpad = sign_lpad(context, lpad);
167                    uw::_Unwind_SetIP(context, maybe_signed_lpad);
168                    uw::_URC_INSTALL_CONTEXT
169                },
170                EHAction::Terminate => uw::_URC_FATAL_PHASE2_ERROR,
171            }
172        }
173    }
174}

176/// personality fn called by [Itanium C++ ABI Exception Handling][itanium-eh]
177///
178/// The personality routine for most non-Windows targets. This will be called by
179/// the unwinding library:
180/// - "In the search phase, the framework repeatedly calls the personality routine, with the
181///   _UA_SEARCH_PHASE flag as described below, first for the current PC and register state, and
182///   then unwinding a frame to a new PC at each step..."
183/// - "If the search phase reports success, the framework restarts in the cleanup phase. Again, it
184///   repeatedly calls the personality routine, with the _UA_CLEANUP_PHASE flag as described below,
185///   first for the current PC and register state, and then unwinding a frame to a new PC at each
186///   step..."i
187///
188/// [itanium-eh]: https://itanium-cxx-abi.github.io/cxx-abi/abi-eh.html
189#[lang = "eh_personality"]
190#[cfg(all(not(feature = "std"), panic = "unwind"))]
191unsafe extern "C" fn rust_eh_personality(
192    version: c_int, actions: uw::_Unwind_Action, exception_class: uw::_Unwind_Exception_Class,
193    exception_object: *mut uw::_Unwind_Exception, context: *mut uw::_Unwind_Context,
194) -> uw::_Unwind_Reason_Code {
195    // SAFETY: the platform support must modify the cfg for the inner fn
196    // if it needs something different than what is currently invoked.
197    unsafe {
198        rust_eh_personality_impl(version, actions, exception_class, exception_object, context)
199    }
200}

208#[cfg(all(not(feature = "std"), panic = "unwind"))]
209unsafe fn find_eh_action(context: *mut uw::_Unwind_Context) -> Result<EHAction, ()> {
210    unsafe {
211        let lsda = uw::_Unwind_GetLanguageSpecificData(context) as *const u8;
212        let mut ip_before_instr: c_int = 0;
213        let ip = uw::_Unwind_GetIPInfo(context, &mut ip_before_instr);
214        let eh_context = EHContext {
215            // The return address points 1 byte past the call instruction,
216            // which could be in the next IP range in LSDA range table.
217            //
218            // `ip = -1` has special meaning, so use wrapping sub to allow for that
219            ip: if ip_before_instr != 0 { ip } else { ip.wrapping_sub(1) },
220            func_start: uw::_Unwind_GetRegionStart(context),
221            get_text_start: &|| uw::_Unwind_GetTextRelBase(context),
222            get_data_start: &|| uw::_Unwind_GetDataRelBase(context),
223        };
224        eh::find_eh_action(lsda, &eh_context)
225    }
