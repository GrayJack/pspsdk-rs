126    unsafe extern "C" fn memset(src: *mut c_void, value: i32, num: SceSize) -> *mut c_void {
127        unsafe {
128            cfg_select! {
129                feature = "kernel" => crate::sys::libc::memset(src.cast(), value, num).cast(),
130                all(not(feature = "kernel"), not(feature = "cfw-api"), feature = "stub-c") => {
131                    crate::sys::usersystemlib::sceKernelMemset(src.cast(), value, num).cast()
132                },
133                all(not(feature = "kernel"), feature = "cfw-api", feature = "stub-c") => {
134                    crate::sys::libc::memset(src, value as i32, num)
135                },
136                _ => {
137                    string_impl::set_bytes(src.cast(), value as u8, num);
138                    src
139                },
140            }
141        }
142    }

178    #[unsafe(no_mangle)]
179    #[cfg(feature = "non-stub-code")]
180    unsafe extern "C" fn memmove(
181        dst: *mut c_void, src: *const c_void, num: SceSize,
182    ) -> *mut c_void {
183        unsafe {
184            cfg_select! {
185                feature = "kernel" => crate::sys::libc::memmove(dst.cast(), src.cast(), num).cast(),
186                all(not(feature = "kernel"), feature = "cfw-api", feature = "stub-c") => {
187                    crate::sys::libc::memmove(dst, src, num)
188                },
189                _ => {
190                    use string_impl::PointersOverlap;
191
192                    let src_byte = src as *const u8;
193                    let dest_byte = dst as *mut u8;
194
195                    // If the pointer overlap, we have to copy backwards
196                    if src_byte.overlaps_with(dest_byte, num as _) {
197                        string_impl::copy_backward(dest_byte, src_byte, num as _);
198                    } else {
199                        string_impl::copy_forward(dest_byte, src_byte, num as _);
200                    }
201
202                    dst
203                },
204            }
205        }
206    }

278/// Enable the home button.
279///
280/// This API does not have destructor support yet. You can manually setup an
281/// exit callback if you need this, see the source code of this function.
282#[cfg(feature = "non-stub-code")]
283pub fn enable_home_button() {
284    use core::{ffi::c_void, ptr};
285    use sys::thread::ThreadAttributes;
286
287    unsafe {
288        unsafe extern "C" fn exit_thread(_args: usize, _argp: *mut c_void) -> SceResult<u32> {
289            #[allow(unreachable_code)]
290            unsafe extern "C" fn exit_callback(
291                _arg1: u32, _arg2: u32, _arg: *mut c_void,
292            ) -> CallbackTermState {
293                crate::rt::cleanup();
294                process::exit_main(0);
295                CallbackTermState::NormalTermination
296            }
297
298            let res = unsafe {
299                sys::thread::sceKernelCreateCallback(
300                    c"exit_callback".as_ptr().cast(),
301                    exit_callback,
302                    ptr::null_mut(),
303                )
304            };
305
306            let raw_res = res.as_inner();
307            let Ok(id) = res.into_result() else {
308                return SceResult::new(raw_res);
309            };
310
311            let _res = sys::loadexec::sceKernelRegisterExitCallback(id);
312            let _res = sys::thread::sceKernelSleepThreadCB();
313
314            SceResult::new(0)
315        }
316
317        // Enable the home button.
318        let id = sys::thread::sceKernelCreateThread(
319            &b"exit_thread\0"[0],
320            exit_thread,
321            0x11,
322            0xFA0,
323            ThreadAttributes::empty(),
324            None,
325        )
326        .into_result();
327
328        let Ok(id) = id else {
329            return;
330        };
331
332        let _res = sys::thread::sceKernelStartThread(id, 0, ptr::null_mut());
333    }
