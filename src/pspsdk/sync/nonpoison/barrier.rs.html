5/// A barrier enables multiple threads to synchronize the beginning
6/// of some computation.
7///
8/// # Examples
9///
10/// ```no_run
11/// use pspsdk::{sync::Barrier, thread};
12///
13/// let n = 10;
14/// let barrier = Barrier::new(n);
15/// thread::scope(|s| {
16///     for _ in 0..n {
17///         // The same messages will be printed together.
18///         // You will NOT see any interleaving.
19///         s.spawn(|| {
20///             println!("before wait");
21///             barrier.wait();
22///             println!("after wait");
23///         });
24///     }
25/// });
26/// ```
27pub struct Barrier {
28    lock: Mutex<BarrierState>,
29    cvar: Condvar,
30    num_threads: usize,
31}
32
33impl RefUnwindSafe for Barrier {}

35// The inner state of a double barrier
36struct BarrierState {
37    count: usize,
38    generation_id: usize,
39}

41/// A `BarrierWaitResult` is returned by [`Barrier::wait()`] when all threads
42/// in the [`Barrier`] have rendezvoused.
43///
44/// # Examples
45///
46/// ```
47/// use pspsdk::sync::Barrier;
48///
49/// let barrier = Barrier::new(1);
50/// let barrier_wait_result = barrier.wait();
51/// ```
52pub struct BarrierWaitResult(bool);

60impl Barrier {
61    /// Creates a new barrier that can block a given number of threads.
62    ///
63    /// A barrier will block all threads which call [`wait()`] until the `n`th thread calls
64    /// [`wait()`], and then wake up all threads at once.
65    ///
66    /// [`wait()`]: Barrier::wait
67    ///
68    /// # Examples
69    ///
70    /// ```
71    /// use pspsdk::sync::nonpoison::Barrier;
72    ///
73    /// let barrier = Barrier::new(10);
74    /// ```
75    #[inline]
76    #[must_use]
77    pub const fn new(n: usize) -> Barrier {
78        Barrier {
79            lock: Mutex::new(BarrierState {
80                count: 0,
81                generation_id: 0,
82            }),
83            cvar: Condvar::new(),
84            num_threads: n,
85        }
86    }
87
88    /// Blocks the current thread until all threads have rendezvoused here.
89    ///
90    /// Barriers are re-usable after all threads have rendezvoused once, and can
91    /// be used continuously.
92    ///
93    /// A single (arbitrary) thread will receive a [`BarrierWaitResult`] that
94    /// returns `true` from [`BarrierWaitResult::is_leader()`] when returning
95    /// from this function, and all other threads will receive a result that
96    /// will return `false` from [`BarrierWaitResult::is_leader()`].
97    ///
98    /// # Examples
99    ///
100    /// ```
101    /// use pspsdk::{sync::Barrier, thread};
102    ///
103    /// let n = 10;
104    /// let barrier = Barrier::new(n);
105    /// thread::scope(|s| {
106    ///     for _ in 0..n {
107    ///         // The same messages will be printed together.
108    ///         // You will NOT see any interleaving.
109    ///         s.spawn(|| {
110    ///             println!("before wait");
111    ///             barrier.wait();
112    ///             println!("after wait");
113    ///         });
114    ///     }
115    /// });
116    /// ```
117    pub fn wait(&self) -> BarrierWaitResult {
118        let mut lock = self.lock.lock();
119        let local_gen = lock.generation_id;
120        lock.count += 1;
121        if lock.count < self.num_threads {
122            self.cvar
123                .wait_while(&mut lock, |state: &mut BarrierState| local_gen == state.generation_id);
124            BarrierWaitResult(false)
125        } else {
126            lock.count = 0;
127            lock.generation_id = lock.generation_id.wrapping_add(1);
128            self.cvar.notify_all();
129            BarrierWaitResult(true)
130        }
131    }
132}
