33/// A mutual exclusion primitive useful for protecting shared data
34///
35/// This mutex will block threads waiting for the lock to become available. The
36/// mutex can be created via a [`new`] constructor. Each mutex has a type parameter
37/// which represents the data that it is protecting. The data can only be accessed
38/// through the RAII guards returned from [`lock`] and [`try_lock`], which
39/// guarantees that the data is only ever accessed when the mutex is locked.
40///
41/// # Poisoning
42///
43/// The mutexes in this module implement a strategy called "poisoning" where a
44/// mutex is considered poisoned whenever a thread panics while holding the
45/// mutex. Once a mutex is poisoned, all other threads are unable to access the
46/// data by default as it is likely tainted (some invariant is not being
47/// upheld).
48///
49/// For a mutex, this means that the [`lock`] and [`try_lock`] methods return a
50/// [`Result`] which indicates whether a mutex has been poisoned or not. Most
51/// usage of a mutex will simply [`unwrap()`] these results, propagating panics
52/// among threads to ensure that a possibly invalid invariant is not witnessed.
53///
54/// A poisoned mutex, however, does not prevent all access to the underlying
55/// data. The [`PoisonError`] type has an [`into_inner`] method which will return
56/// the guard that would have otherwise been returned on a successful lock. This
57/// allows access to the data, despite the lock being poisoned.
58///
59/// [`new`]: Self::new
60/// [`lock`]: Self::lock
61/// [`try_lock`]: Self::try_lock
62/// [`unwrap()`]: Result::unwrap
63/// [`PoisonError`]: super::PoisonError
64/// [`into_inner`]: super::PoisonError::into_inner
65///
66/// # Examples
67///
68/// ```
69/// use std::{
70///     sync::{mpsc::channel, Arc},
71///     thread,
72/// };
73///
74/// use pspsdk::sync::poison::Mutex;
75///
76/// const N: usize = 10;
77///
78/// // Spawn a few threads to increment a shared variable (non-atomically), and
79/// // let the main thread know once all increments are done.
80/// //
81/// // Here we're using an Arc to share memory among threads, and the data inside
82/// // the Arc is protected with a mutex.
83/// let data = Arc::new(Mutex::new(0));
84///
85/// let (tx, rx) = channel();
86/// for _ in 0..N {
87///     let (data, tx) = (Arc::clone(&data), tx.clone());
88///     thread::spawn(move || {
89///         // The shared state can only be accessed once the lock is held.
90///         // Our non-atomic increment is safe because we're the only thread
91///         // which can access the shared state when the lock is held.
92///         //
93///         // We unwrap() the return value to assert that we are not expecting
94///         // threads to ever fail while holding the lock.
95///         let mut data = data.lock();
96///         *data += 1;
97///         if *data == N {
98///             tx.send(()).unwrap();
99///         }
100///         // the lock is unlocked here when `data` goes out of scope.
101///     });
102/// }
103///
104/// rx.recv().unwrap();
105/// ```
106///
107/// To recover from a poisoned mutex:
108///
109/// ```
110/// use std::{sync::Arc, thread};
111///
112/// use pspsdk::sync::poison::Mutex;
113///
114/// let lock = Arc::new(Mutex::new(0_u32));
115/// let lock2 = Arc::clone(&lock);
116///
117/// let _ = thread::spawn(move || -> () {
118///     // This thread will acquire the mutex first, unwrapping the result of
119///     // `lock` because the lock has not been poisoned.
120///     let _guard = lock2.lock();
121///
122///     // This panic while holding the lock (`_guard` is in scope) will poison
123///     // the mutex.
124///     panic!();
125/// })
126/// .join();
127///
128/// // The lock is poisoned by this point, but the returned result can be
129/// // pattern matched on to return the underlying guard on both branches.
130/// let mut guard = match lock.lock_checked() {
131///     Ok(guard) => guard,
132///     Err(poisoned) => poisoned.into_inner(),
133/// };
134///
135/// *guard += 1;
136/// ```
137///
138/// To unlock a mutex guard sooner than the end of the enclosing scope,
139/// either create an inner scope or drop the guard manually.
140///
141/// ```
142/// use std::{sync::Arc, thread};
143///
144/// use pspsdk::sync::poison::Mutex;
145///
146/// const N: usize = 3;
147///
148/// let data_mutex = Arc::new(Mutex::new(vec![1, 2, 3, 4]));
149/// let res_mutex = Arc::new(Mutex::new(0));
150///
151/// let mut threads = Vec::with_capacity(N);
152/// (0..N).for_each(|_| {
153///     let data_mutex_clone = Arc::clone(&data_mutex);
154///     let res_mutex_clone = Arc::clone(&res_mutex);
155///
156///     threads.push(thread::spawn(move || {
157///         // Here we use a block to limit the lifetime of the lock guard.
158///         let result = {
159///             let mut data = data_mutex_clone.lock();
160///             // This is the result of some important and long-ish work.
161///             let result = data.iter().fold(0, |acc, x| acc + x * 2);
162///             data.push(result);
163///             result
164///             // The mutex guard gets dropped here, together with any other values
165///             // created in the critical section.
166///         };
167///         // The guard created here is a temporary dropped at the end of the statement, i.e.
168///         // the lock would not remain being held even if the thread did some additional work.
169///         *res_mutex_clone.lock() += result;
170///     }));
171/// });
172///
173/// let mut data = data_mutex.lock();
174/// // This is the result of some important and long-ish work.
175/// let result = data.iter().fold(0, |acc, x| acc + x * 2);
176/// data.push(result);
177/// // We drop the `data` explicitly because it's not necessary anymore and the
178/// // thread still has work to do. This allows other threads to start working on
179/// // the data immediately, without waiting for the rest of the unrelated work
180/// // to be done here.
181/// //
182/// // It's even more important here than in the threads because we `.join` the
183/// // threads after that. If we had not dropped the mutex guard, a thread could
184/// // be waiting forever for it, causing a deadlock.
185/// // As in the threads, a block could have been used instead of calling the
186/// // `drop` function.
187/// drop(data);
188/// // Here the mutex guard is not assigned to a variable and so, even if the
189/// // scope does not end after this line, the mutex is still released: there is
190/// // no deadlock.
191/// *res_mutex.lock() += result;
192///
193/// threads
194///     .into_iter()
195///     .for_each(|thread| thread.join().expect("The thread creating or execution failed !"));
196///
197/// assert_eq!(*res_mutex.lock(), 800);
198/// ```
199pub struct Mutex<T: ?Sized, M = DefaultMutex> {
200    inner: M,
201    poison: poison::Flag,
202    data: UnsafeCell<T>,
203}

213/// An RAII implementation of a "scoped lock" of a mutex. When this structure is
214/// dropped (falls out of scope), the lock will be unlocked.
215///
216/// The data protected by the mutex can be accessed through this guard via its
217/// [`Deref`] and [`DerefMut`] implementations.
218///
219/// This structure is created by the [`lock`] and [`try_lock`] methods on
220/// [`Mutex`].
221///
222/// [`lock`]: Mutex::lock
223/// [`try_lock`]: Mutex::try_lock
224#[must_use = "if unused the Mutex will immediately unlock"]
225#[clippy::has_significant_drop]
226pub struct MutexGuard<'a, T: ?Sized + 'a, M: RawMutex = DefaultMutex> {
227    lock: &'a Mutex<T, M>,
228    poison: poison::Guard,
229}

234/// An RAII mutex guard returned by `PoisonMutexGuard::map`, which can point to a
235/// subfield of the protected data. When this structure is dropped (falls out
236/// of scope), the lock will be unlocked.
237///
238/// The main difference between `MappedMutexGuard` and [`MutexGuard`] is that the
239/// former cannot be used with [`Condvar`], since that
240/// could introduce soundness issues if the locked object is modified by another
241/// thread while the `Mutex` is unlocked.
242///
243/// The data protected by the mutex can be accessed through this guard via its
244/// [`Deref`] and [`DerefMut`] implementations.
245///
246/// This structure is created by the [`map`] and [`try_map`] methods on
247/// [`MutexGuard`].
248///
249/// [`map`]: MutexGuard::map
250/// [`try_map`]: MutexGuard::try_map
251/// [`Condvar`]: crate::sync::poison::Condvar
252#[must_use = "if unused the Mutex will immediately unlock"]
253#[clippy::has_significant_drop]
254pub struct MappedMutexGuard<'a, T: ?Sized + 'a, M: RawMutex = DefaultMutex> {
255    // NB: we use a pointer instead of `&'a mut T` to avoid `noalias` violations, because a
256    // `MappedMutexGuard` argument doesn't hold uniqueness for its whole scope, only until it
257    // drops. `NonNull` is covariant over `T`, so we add a `PhantomData<&'a mut T>` field
258    // below for the correct variance over `T` (invariance).
259    data: NonNull<T>,
260    inner: &'a M,
261    poison_flag: &'a poison::Flag,
262    poison: poison::Guard,
263    _variance: PhantomData<&'a mut T>,
264}
265
266impl<T: ?Sized, M: RawMutex> !Send for MappedMutexGuard<'_, T, M> {}
267unsafe impl<T: ?Sized + Sync, M: RawMutex> Sync for MappedMutexGuard<'_, T, M> {}

290    /// Creates a new mutex in an unlocked state ready for use using the specified mutex.
291    #[inline]
292    pub const fn new_with(value: T, raw_mutex: M) -> Self {
293        Self {
294            inner: raw_mutex,
295            poison: poison::Flag::new(),
296            data: UnsafeCell::new(value),
297        }

300    /// Returns the contained value by cloning it.
301    ///
302    /// # Errors
303    ///
304    /// If another user of this mutex panicked while holding the mutex, then
305    /// this call will return an error instead.
306    ///
307    /// # Examples
308    ///
309    /// ```
310    /// use pspsdk::sync::poison::Mutex;
311    ///
312    /// let mut mutex = Mutex::new(7);
313    ///
314    /// assert_eq!(mutex.get_cloned().unwrap(), 7);
315    /// ```
316    pub fn get_cloned(&self) -> Result<T, PoisonError<()>>
317    where
318        T: Clone,
319    {
320        match self.lock_checked() {
321            Ok(guard) => Ok((*guard).clone()),
322            Err(_) => Err(PoisonError::new(())),
323        }
324    }
325
326    /// Sets the contained value.
327    ///
328    /// # Errors
329    ///
330    /// If another user of this mutex panicked while holding the mutex, then
331    /// this call will return an error containing the provided `value` instead.
332    ///
333    /// # Examples
334    ///
335    /// ```
336    /// use pspsdk::sync::poison::Mutex;
337    ///
338    /// let mut mutex = Mutex::new(7);
339    ///
340    /// assert_eq!(mutex.get_cloned().unwrap(), 7);
341    /// mutex.set(11).unwrap();
342    /// assert_eq!(mutex.get_cloned().unwrap(), 11);
343    /// ```
344    pub fn set(&self, value: T) -> Result<(), PoisonError<T>> {
345        if mem::needs_drop::<T>() {
346            // If the contained value has non-trivial destructor, we
347            // call that destructor after the lock being released.
348            self.replace(value).map(drop)
349        } else {
350            match self.lock_checked() {
351                Ok(mut guard) => {
352                    *guard = value;
353
354                    Ok(())
355                },
356                Err(_) => Err(PoisonError::new(value)),
357            }
358        }
359    }
360
361    /// Replaces the contained value with `value`, and returns the old contained value.
362    ///
363    /// # Errors
364    ///
365    /// If another user of this mutex panicked while holding the mutex, then
366    /// this call will return an error containing the provided `value` instead.
367    ///
368    /// # Examples
369    ///
370    /// ```
371    /// use pspsdk::sync::poison::Mutex;
372    ///
373    /// let mut mutex = Mutex::new(7);
374    ///
375    /// assert_eq!(mutex.replace(11).unwrap(), 7);
376    /// assert_eq!(mutex.get_cloned().unwrap(), 11);
377    /// ```
378    pub fn replace(&self, value: T) -> LockResult<T> {
379        match self.lock_checked() {
380            Ok(mut guard) => Ok(mem::replace(&mut *guard, value)),
381            Err(_) => Err(PoisonError::new(value)),
382        }
383    }
384}

386impl<T: ?Sized, M: RawMutex> Mutex<T, M> {
387    /// Acquires a mutex, blocking the current thread until it is able to do so.
388    ///
389    /// This function will block the local thread until it is available to acquire
390    /// the mutex. Upon returning, the thread is the only thread with the lock
391    /// held. An RAII guard is returned to allow scoped unlock of the lock. When
392    /// the guard goes out of scope, the mutex will be unlocked.
393    ///
394    /// The exact behavior on locking a mutex in the thread which already holds
395    /// the lock is left unspecified. However, this function will not return on
396    /// the second call (it might panic or deadlock, for example).
397    ///
398    /// # Panics
399    ///
400    /// This function might panic when called if the lock is already held by
401    /// the current thread or the lock is in a poisoned state.
402    ///
403    /// # Examples
404    ///
405    /// ```
406    /// use std::{sync::Arc, thread};
407    ///
408    /// use pspsdk::sync::poison::Mutex;
409    ///
410    /// let mutex = Arc::new(Mutex::new(0));
411    /// let c_mutex = Arc::clone(&mutex);
412    ///
413    /// thread::spawn(move || {
414    ///     *c_mutex.lock() = 10;
415    /// })
416    /// .join()
417    /// .expect("thread::spawn failed");
418    /// assert_eq!(*mutex.lock(), 10);
419    /// ```
420    #[track_caller]
421    pub fn lock(&self) -> MutexGuard<'_, T, M> {
422        self.inner.lock();
423        match unsafe { MutexGuard::new(self) } {
424            Ok(guard) => guard,
425            Err(err) => panic!("{err}"),
426        }
427    }
428
429    /// Attempts to acquire this lock.
430    ///
431    /// If the lock could not be acquired at this time, then [`Err`] is returned.
432    /// Otherwise, an RAII guard is returned. The lock will be unlocked when the
433    /// guard is dropped.
434    ///
435    /// This function does not block.
436    ///
437    /// # Errors
438    ///
439    /// If another user of this mutex panicked while holding the mutex, then
440    /// this call will return the [`Poisoned`] error if the mutex would
441    /// otherwise be acquired. An acquired lock guard will be contained
442    /// in the returned error.
443    ///
444    /// If the mutex could not be acquired because it is already locked, then
445    /// this call will return the [`WouldBlock`] error.
446    ///
447    /// [`Poisoned`]: TryLockError::Poisoned
448    /// [`WouldBlock`]: TryLockError::WouldBlock
449    ///
450    /// # Examples
451    ///
452    /// ```
453    /// use std::{sync::Arc, thread};
454    ///
455    /// use pspsdk::sync::poison::Mutex;
456    ///
457    /// let mutex = Arc::new(Mutex::new(0));
458    /// let c_mutex = Arc::clone(&mutex);
459    ///
460    /// thread::spawn(move || {
461    ///     let mut lock = c_mutex.try_lock();
462    ///     if let Ok(ref mut mutex) = lock {
463    ///         **mutex = 10;
464    ///     } else {
465    ///         println!("try_lock failed");
466    ///     }
467    /// })
468    /// .join()
469    /// .expect("thread::spawn failed");
470    /// assert_eq!(*mutex.lock(), 10);
471    /// ```
472    pub fn try_lock(&self) -> TryLockResult<MutexGuard<'_, T, M>> {
473        if self.inner.try_lock() {
474            Ok(unsafe { MutexGuard::new(self)? })
475        } else {
476            Err(TryLockError::WouldBlock)
477        }
478    }

480    /// Acquires a mutex, blocking the current thread until it is able to do so.
481    ///
482    /// This function will block the local thread until it is available to acquire
483    /// the mutex. Upon returning, the thread is the only thread with the lock
484    /// held. An RAII guard is returned to allow scoped unlock of the lock. When
485    /// the guard goes out of scope, the mutex will be unlocked.
486    ///
487    /// The exact behavior on locking a mutex in the thread which already holds
488    /// the lock is left unspecified. However, this function will not return on
489    /// the second call (it might panic or deadlock, for example).
490    ///
491    /// # Errors
492    ///
493    /// If another user of this mutex panicked while holding the mutex, then
494    /// this call will return an error once the mutex is acquired. The acquired
495    /// mutex guard will be contained in the returned error.
496    ///
497    /// # Panics
498    ///
499    /// This function might panic when called if the lock is already held by
500    /// the current thread.
501    ///
502    /// # Examples
503    ///
504    /// ```
505    /// use std::{sync::Arc, thread};
506    ///
507    /// use pspsdk::sync::poison::Mutex;
508    ///
509    /// let mutex = Arc::new(Mutex::new(0));
510    /// let c_mutex = Arc::clone(&mutex);
511    ///
512    /// thread::spawn(move || {
513    ///     *c_mutex.lock_checked().unwrap() = 10;
514    /// })
515    /// .join()
516    /// .expect("thread::spawn failed");
517    /// assert_eq!(*mutex.lock(), 10);
518    /// ```
519    pub fn lock_checked(&self) -> LockResult<MutexGuard<'_, T, M>> {
520        self.inner.lock();
521        unsafe { MutexGuard::new(self) }
522    }

524    /// Same as [`lock`](Self::lock), but ignoring poison state instead of panicking.
525    ///
526    /// # Panics
527    ///
528    /// This function might panic when called if the lock is already held by
529    /// the current thread.
530    pub fn lock_unchecked(&self) -> MutexGuard<'_, T, M> {
531        self.inner.lock();
532        match unsafe { MutexGuard::new(self) } {
533            Ok(guard) => guard,
534            Err(err) => err.into_inner(),
535        }
536    }

538    /// Same as [`try_lock`](Self::try_lock), but ignoring poison state.
539    ///
540    /// # Errors
541    ///
542    /// If the mutex could not be acquired because it is already locked, then
543    /// this call will return the [`WouldBlock`] error.
544    ///
545    /// [`WouldBlock`]: TryLockError::WouldBlock
546    pub fn try_unchecked(&self) -> TryLockResult<MutexGuard<'_, T, M>> {
547        match self.try_lock() {
548            Ok(guard) => Ok(guard),
549            Err(TryLockError::Poisoned(err)) => Ok(err.into_inner()),
550            Err(err) => Err(err),
551        }
552    }

554    /// Determines whether the mutex is poisoned.
555    ///
556    /// If another thread is active, the mutex can still become poisoned at any
557    /// time. You should not trust a `false` value for program correctness
558    /// without additional synchronization.
559    ///
560    /// # Examples
561    ///
562    /// ```
563    /// use std::{sync::Arc, thread};
564    ///
565    /// use pspsdk::sync::poison::Mutex;
566    ///
567    /// let mutex = Arc::new(Mutex::new(0));
568    /// let c_mutex = Arc::clone(&mutex);
569    ///
570    /// let _ = thread::spawn(move || {
571    ///     let _lock = c_mutex.lock();
572    ///     panic!(); // the mutex gets poisoned
573    /// })
574    /// .join();
575    /// assert_eq!(mutex.is_poisoned(), true);
576    /// ```
577    #[inline]
578    pub fn is_poisoned(&self) -> bool {
579        self.poison.get()
580    }

582    /// Clear the poisoned state from a mutex.
583    ///
584    /// If the mutex is poisoned, it will remain poisoned until this function is called. This
585    /// allows recovering from a poisoned state and marking that it has recovered. For example, if
586    /// the value is overwritten by a known-good value, then the mutex can be marked as
587    /// un-poisoned. Or possibly, the value could be inspected to determine if it is in a
588    /// consistent state, and if so the poison is removed.
589    ///
590    /// # Examples
591    ///
592    /// ```
593    /// use std::{sync::Arc, thread};
594    ///
595    /// use pspsdk::sync::poison::Mutex;
596    ///
597    /// let mutex = Arc::new(Mutex::new(0));
598    /// let c_mutex = Arc::clone(&mutex);
599    ///
600    /// let _ = thread::spawn(move || {
601    ///     let _lock = c_mutex.lock();
602    ///     panic!(); // the mutex gets poisoned
603    /// })
604    /// .join();
605    ///
606    /// assert_eq!(mutex.is_poisoned(), true);
607    /// let x = mutex.lock_checked().unwrap_or_else(|mut e| {
608    ///     **e.get_mut() = 1;
609    ///     mutex.clear_poison();
610    ///     e.into_inner()
611    /// });
612    /// assert_eq!(mutex.is_poisoned(), false);
613    /// assert_eq!(*x, 1);
614    /// ```
615    #[inline]
616    pub fn clear_poison(&self) {
617        self.poison.clear();
618    }

620    /// Consumes this mutex, returning the underlying data.
621    ///
622    /// # Panics
623    ///
624    /// If another user of this mutex panicked while holding the mutex, then
625    /// this call will panic instead.
626    ///
627    /// # Examples
628    ///
629    /// ```
630    /// use pspsdk::sync::poison::Mutex;
631    ///
632    /// let mutex = Mutex::new(0);
633    /// assert_eq!(mutex.into_inner(), 0);
634    /// ```
635    #[track_caller]
636    pub fn into_inner(self) -> T
637    where
638        T: Sized,
639    {
640        let data = self.data.into_inner();
641        match poison::map_result(self.poison.borrow(), |()| data) {
642            Ok(data) => data,
643            Err(err) => panic!("{err}"),
644        }
645    }

647    /// Consumes this mutex, returning the underlying data, explicitly cheking poison state.
648    ///
649    /// # Errors
650    ///
651    /// If another user of this mutex panicked while holding the mutex, then
652    /// this call will return an error containing the the underlying data
653    /// instead.
654    ///
655    /// # Examples
656    ///
657    /// ```
658    /// use pspsdk::sync::poison::Mutex;
659    ///
660    /// let mutex = Mutex::new(0);
661    /// assert_eq!(mutex.into_inner_checked().unwrap(), 0);
662    /// ```
663    pub fn into_inner_checked(self) -> LockResult<T>
664    where
665        T: Sized,
666    {
667        let data = self.data.into_inner();
668        poison::map_result(self.poison.borrow(), |()| data)
669    }

671    /// Returns a mutable reference to the underlying data.
672    ///
673    /// Since this call borrows the `Mutex` mutably, no actual locking needs to
674    /// take place -- the mutable borrow statically guarantees no locks exist.
675    ///
676    /// # Panics
677    ///
678    /// If another user of this mutex panicked while holding the mutex, then
679    /// this call will panic instead.
680    ///
681    /// # Examples
682    ///
683    /// ```
684    /// use pspsdk::sync::poison::Mutex;
685    ///
686    /// let mut mutex = Mutex::new(0);
687    /// *mutex.get_mut() = 10;
688    /// assert_eq!(*mutex.lock(), 10);
689    /// ```
690    #[track_caller]
691    pub fn get_mut(&mut self) -> &mut T {
692        let data = self.data.get_mut();
693        match poison::map_result(self.poison.borrow(), |()| data) {
694            Ok(data) => data,
695            Err(err) => panic!("{err}"),
696        }
697    }

699    /// Returns a mutable reference to the underlying data, explicitly cheking poison state.
700    ///
701    /// Since this call borrows the `Mutex` mutably, no actual locking needs to
702    /// take place -- the mutable borrow statically guarantees no locks exist.
703    ///
704    /// # Errors
705    ///
706    /// If another user of this mutex panicked while holding the mutex, then
707    /// this call will return an error containing a mutable reference to the
708    /// underlying data instead.
709    ///
710    /// # Examples
711    /// ```
712    /// use pspsdk::sync::poison::Mutex;
713    ///
714    /// let mut mutex = Mutex::new(0);
715    /// *mutex.get_mut_checked().unwrap() = 10;
716    /// assert_eq!(*mutex.lock(), 10);
717    /// ```
718    pub fn get_mut_checked(&mut self) -> LockResult<&mut T> {
719        let data = self.data.get_mut();
720        poison::map_result(self.poison.borrow(), |()| data)
721    }

723    /// Consumes this mutex ignoring the poison state, returning the underlying data.
724    ///
725    /// # Examples
726    ///
727    /// ```
728    /// use pspsdk::sync::poison::Mutex;
729    ///
730    /// let mutex = Mutex::new(0);
731    /// assert_eq!(mutex.into_inner_unchecked(), 0);
732    /// ```
733    pub fn into_inner_unchecked(self) -> T
734    where
735        T: Sized,
736    {
737        self.data.into_inner()
738    }

740    /// Returns a mutable reference to the underlying data, ignoring poison state.
741    ///
742    /// Since this call borrows the `Mutex` mutably, no actual locking needs to
743    /// take place -- the mutable borrow statically guarantees no locks exist.
744    ///
745    /// # Examples
746    ///
747    /// ```
748    /// use pspsdk::sync::poison::Mutex;

752    /// assert_eq!(*mutex.lock(), 10);
753    /// ```
754    pub fn get_mut_unchecked(&mut self) -> &mut T {
755        self.data.get_mut()
756    }
757
758    /// Returns a raw pointer to the underlying data.
759    ///
760    /// The returned pointer is always non-null and properly aligned, but it is
761    /// the user's responsibility to ensure that any reads and writes through it
762    /// are properly synchronized to avoid data races, and that it is not read
763    /// or written through after the mutex is dropped.
764    pub const fn data_ptr(&self) -> *mut T {
765        self.data.get()
766    }

823        crate::println!("GOT HERE 1");
824        let mut d = f.debug_struct("Mutex");
825        match self.try_lock() {
826            Ok(guard) => {
827                crate::println!("GOT HERE 2");

831                d.field("data", &&**err.get_ref());
832            },

834                d.field("data", &format_args!("<locked>"));
835            },
836        }
837        d.field("poisoned", &self.poison.get());
838        d.finish_non_exhaustive()
839    }

911impl<'a, T: ?Sized, M: RawMutex> MutexGuard<'a, T, M> {
912    /// Makes a [`MappedMutexGuard`] for a component of the borrowed data, e.g.
913    /// an enum variant.
914    ///
915    /// The `Mutex` is already locked, so this cannot fail.
916    ///
917    /// This is an associated function that needs to be used as
918    /// `MutexGuard::map(...)`. A method would interfere with methods of the
919    /// same name on the contents of the `MutexGuard` used through `Deref`.
920    pub fn map<U, F>(orig: Self, f: F) -> MappedMutexGuard<'a, U, M>
921    where
922        F: FnOnce(&mut T) -> &mut U,
923        U: ?Sized,
924    {
925        // SAFETY: the conditions of `MutexGuard::new` were satisfied when the original guard
926        // was created, and have been upheld throughout `map` and/or `try_map`.
927        // The signature of the closure guarantees that it will not "leak" the lifetime of the
928        // reference passed to it. If the closure panics, the guard will be dropped.
929        let data = NonNull::from(f(unsafe { &mut *orig.lock.data.get() }));
930        let orig = ManuallyDrop::new(orig);
931        MappedMutexGuard {
932            data,
933            inner: &orig.lock.inner,
934            poison_flag: &orig.lock.poison,
935            poison: orig.poison.clone(),
936            _variance: PhantomData,
937        }
938    }
939
940    /// Makes a [`MappedMutexGuard`] for a component of the borrowed data. The
941    /// original guard is returned as an `Err(...)` if the closure returns
942    /// `None`.
943    ///
944    /// The `Mutex` is already locked, so this cannot fail.
945    ///
946    /// This is an associated function that needs to be used as
947    /// `MutexGuard::try_map(...)`. A method would interfere with methods of the
948    /// same name on the contents of the `MutexGuard` used through `Deref`.
949    #[doc(alias = "filter_map")]
950    pub fn try_map<U, F>(orig: Self, f: F) -> Result<MappedMutexGuard<'a, U, M>, Self>
951    where
952        F: FnOnce(&mut T) -> Option<&mut U>,
953        U: ?Sized,
954    {
955        // SAFETY: the conditions of `MutexGuard::new` were satisfied when the original guard
956        // was created, and have been upheld throughout `map` and/or `try_map`.
957        // The signature of the closure guarantees that it will not "leak" the lifetime of the
958        // reference passed to it. If the closure panics, the guard will be dropped.
959        match f(unsafe { &mut *orig.lock.data.get() }) {
960            Some(data) => {
961                let data = NonNull::from(data);
962                let orig = ManuallyDrop::new(orig);
963                Ok(MappedMutexGuard {
964                    data,
965                    inner: &orig.lock.inner,
966                    poison_flag: &orig.lock.poison,
967                    poison: orig.poison.clone(),
968                    _variance: PhantomData,
969                })
970            },
971            None => Err(orig),
972        }
973    }
974}
