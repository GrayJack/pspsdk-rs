9/// Define basic operations for a mutex.
10///
11/// A type that implements this can be used with the [`Mutex`] type to have a mutex that functions
12/// similarly as the Rust standard library mutex.
13///
14/// [`Mutex`]: crate::sync::poison::Mutex
15pub trait RawMutex: Sealed {
16    // HACK: using constants while we can't implement const functions on Traits
17    /// A constant new to overcome the limitations of traits not having const fn yet on trits
18    const NEW: Self;
19
20    /// Locks the mutex or panics if the lock is already held by the current thread.
21    fn lock(&self);
22
23    /// Tries to acquire this mutex without blocking, returning `true` if the lock was successfull
24    /// and `false` otherwise.
25    fn try_lock(&self) -> bool;
26
27    /// Unlocks this mutex.
28    ///
29    /// # Safety
30    ///
31    /// This method may only be called if the mutex is held in the current context, i.e. it must
32    /// be paired with a successful call to [`lock`], [`try_lock`], [`try_lock_for`] or
33    /// [`try_lock_until`].
34    ///
35    /// [`lock`]: RawMutex::lock
36    /// [`try_lock`]: RawMutex::try_lock
37    /// [`try_lock_for`]: RawMutexTimed::try_lock_for
38    /// [`try_lock_until`]: RawMutexTimed::try_lock_until
39    unsafe fn unlock(&self);
40
41    fn is_locked(&self) -> bool {
42        let was_acquired = self.try_lock();
43        // we need to unlock now if it was acquired
44        if was_acquired {
45            // SAFETY: The lock was acquired with success above
46            unsafe { self.unlock() };
47        }
48        !was_acquired
49    }
50}

52/// Define timed operation on a mutex.
53pub trait RawMutexTimed: RawMutex {
54    /// Attempts to acquire this lock for a `timeout` duration.
55    fn try_lock_for(&self, timeout: Duration) -> bool;
56
57    /// Attempts to acquire this lock until a timeout is reached.
58    #[inline]
59    fn try_lock_until(&self, timeout: Instant) -> bool {
60        let now = Instant::now();
61        let duration = timeout.duration_since(now);
62
63        // if zero, now is later than timeout, so we past the time to try
64        if duration.is_zero() {
65            false
66        } else {
67            self.try_lock_for(duration)
68        }
69    }
70}

72/// Define basic operations for a ReadWrite Lock.
73///
74/// A type that implements this can be used with the [`RwLock`] type to have a lock that functions
75/// similarly as the Rust standard library rwlock.
76///
77/// [`RwLock`]: crate::sync::poison::RwLock
78pub trait RawRwLock: Sealed {
79    const NEW: Self;
80
81    /// Acquires a read lock, blocking the current thread until it is able to do so.
82    fn read(&self);
83
84    /// Attempts to acquire a read lock without blocking.
85    fn try_read(&self) -> bool;
86
87    /// Acquires an write lock, blocking the current thread until it is able to do so.
88    fn write(&self);
89
90    /// Attempts to acquire an write lock without blocking.
91    fn try_write(&self) -> bool;
92
93    /// Releases a read lock.
94    ///
95    /// # Safety
96    ///
97    /// This method may only be called if a read lock is held in the current context.
98    unsafe fn read_unlock(&self);
99
100    /// Releases an write lock.
101    ///
102    /// # Safety
103    ///
104    /// This method may only be called if an write lock is held in the current context.
105    unsafe fn write_unlock(&self);
106
107    /// Atomically downgrades an write lock into a read lock without
108    /// allowing any thread to take an write lock in the meantime.
109    ///
110    /// # Safety
111    ///
112    /// This method may only be called if an write lock is held in the current context.
113    unsafe fn downgrade(&self);
114
115    /// Checks if this `RwLock` is currently locked in any way.
116    #[inline]
117    fn is_locked(&self) -> bool {
118        let acquired_lock = self.try_write();
119        if acquired_lock {
120            // Safety: A lock was successfully acquired above.
121            unsafe {
122                self.write_unlock();
123            }
124        }
125        !acquired_lock
126    }
127
128    /// Check if this `RwLock` is currently exclusively locked (write locked).
129    fn is_locked_exclusive(&self) -> bool {
130        let acquired_lock = self.try_read();
131        if acquired_lock {
132            // Safety: A shared lock was successfully acquired above.
133            unsafe {
134                self.write_unlock();
135            }
136        }
137        !acquired_lock
138    }
139}
