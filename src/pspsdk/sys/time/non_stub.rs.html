8// PSP epoch is 0001-01-01 00:00:00 UTC.
9// Unix epoch is 1970-01-01 00:00:00 UTC.
10// Offset in seconds: 62,135,596,800
11// PSP tick resolution is 1,000,000 ticks/sec (microsecond precision).
12const PSP_TO_UNIX_EPOCH_SECS: u64 = 62_135_596_800;
13
14pub(crate) const UNIX_EPOCH: SystemTime = SystemTime(Duration::from_secs(PSP_TO_UNIX_EPOCH_SECS));
15
16// Based on sceRtcGetCurrentTick, i.e. ticks since PSP epoch which is 0001-01-01.
17#[derive(Copy, Clone, PartialEq, Eq, PartialOrd, Ord, Debug, Hash)]
18pub(crate) struct SystemTime(Duration);
19
20impl SystemTime {
21    pub const MAX: Self = Self(Duration::MAX);
22    pub const MIN: Self = Self(Duration::ZERO);
23
24    #[allow(clippy::manual_checked_ops)]
25    pub fn now() -> Self {
26        let mut tick = 0;
27        let res = sceRtcGetCurrentTick(&mut tick);
28        debug_assert!(res.is_ok(), "failed to get current tick: {:#X}", res.as_inner());
29        let resolution = sceRtcGetTickResolution() as u64;
30
31        if resolution > 0 {
32            let seconds = tick / resolution;
33            let remaining = tick % resolution;
34            let nanosecs = (remaining * 1_000_000_000) / resolution;
35            Self(Duration::new(seconds, nanosecs as u32))
36        } else {
37            // Assume microseconds resolution as fallback
38            Self(Duration::from_micros(tick))
39        }
40    }
41
42    pub fn sub_time(&self, other: &SystemTime) -> Result<Duration, Duration> {
43        self.0.checked_sub(other.0).ok_or_else(|| other.0 - self.0)
44    }
45
46    pub fn checked_add_duration(&self, other: &Duration) -> Option<SystemTime> {
47        Some(SystemTime(self.0.checked_add(*other)?))
48    }
49
50    pub fn checked_sub_duration(&self, other: &Duration) -> Option<SystemTime> {
51        Some(SystemTime(self.0.checked_sub(*other)?))
52    }
53}

55// Monotonic clock based on sceKernelGetSystemTimeWide() -- microseconds since boot.
56#[derive(Copy, Clone, PartialEq, Eq, PartialOrd, Ord, Debug, Hash)]
57pub(crate) struct Instant(Duration);
58
59impl Instant {
60    pub fn now() -> Self {
61        let micros = sceKernelGetSystemTimeWide();
62        Self(Duration::from_micros(micros))
63    }
64
65    pub fn checked_sub_instant(&self, other: &Instant) -> Option<Duration> {
66        self.0.checked_sub(other.0)
67    }
68
69    pub fn checked_add_duration(&self, other: &Duration) -> Option<Self> {
70        self.0.checked_add(*other).map(Self)
71    }
72
73    pub fn checked_sub_duration(&self, other: &Duration) -> Option<Self> {
74        self.0.checked_sub(*other).map(Self)
75    }
