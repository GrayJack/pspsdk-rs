5/// A "function" stub.
6///
7/// This is a very dirty trick for LTO. Essentially, the PSP OS takes the
8/// address of a stub and inserts 2 instructions (8 bytes) which are `jr $ra`
9/// and `syscall xxx`. Traditionally, the C/C++ toolchain stores the initial
10/// data here as just an empty function that immediately returns (8 bytes,
11/// `jr $ra; nop`). However, we use it to store 2 references: to the NID and to
12/// the library stub.
13///
14/// This results in LTO builds compiling in the NID and library stubs whenever a
15/// function is called, as this struct references them. Thus, this struct
16/// definition creates a dependency between the function call, and the NID + lib
17/// stub. As mentioned earlier, these two addresses (8 bytes) are replaced with
18/// two instructions (also 8 bytes) at runtime, so they are not actually called
19/// as a function. With this method, nothing has to be marked `#[used]`, so LLVM
20/// can automatically remove unreferenced NIDs and library stubs during LTO.
21/// Compiling with LTO then only links the functions that are called, and no
22/// more.
23#[derive(Copy, Clone)]
24pub(crate) struct Stub {
25    // These are never read, but need to be written into as static items.
26    #[allow(dead_code)]
27    pub(crate) lib_addr: &'static StubLibraryEntry,
28    #[allow(dead_code)]
29    pub(crate) nid_addr: &'static u32,
30}

32/// Calculate the padded length for a library name.
33///
34/// The name is padded on the end with zeroes. Must be at least one and a
35/// multiple of 4.
36pub const fn lib_name_bytes_len(name: &str) -> usize {
37    let name_len = name.len();
38    name_len + (4 - name_len % 4)
39}

41/// Convert a library name to a byte array.
42///
43/// This is intended to be used with `lib_name_bytes_len`.
44pub const fn lib_name_bytes<const T: usize>(name: &str) -> [u8; T] {
45    let mut buf = [0; T];
46
47    let name_bytes = name.as_bytes();
48    let mut i = 0;
49
50    while i < name_bytes.len() {
51        buf[i] = name_bytes[i];
52        i += 1;
53    }
54
55    buf
56}
