9/// Audio codec state and parameter structure.
10#[repr(C)]
11#[derive(Debug)]
12#[doc(alias("SceAudiocodecCodec"))]
13pub struct CodecInfo {
14    unk0: i32,
15    unk1: i32,
16    /// Error value if something gone wrong on coded decoding.
17    error: i32,
18    /// The EDRAM address used by the codec.
19    edram_addr: *mut c_void,
20    /// The amount of EDRAM required by the codec, in bytes.
21    needed_mem: u32,
22    unk2: i32,
23    /// The input audio buffer.
24    in_buffer: *mut u8,
25    /// The number of samples read from the input.
26    read_sample: u32,
27    /// The output decoded audio buffer.
28    out_buffer: *mut u8,
29    /// The number of samples produced by decoding.
30    decoded_sample: u32,
31
32    // Note: this part is probably completely different depending on the codec. This should be
33    // cleaned up.
34    /// Proabably samplerate
35    unk3: i32,
36    unk4: i32,
37    unk5: i32,
38    unk6: i32,
39    unk7: i32,
40    unk8: i32,
41    unk9: i32,
42    unk10: i32,
43    unk11: i32,
44    unk12: i32,
45    unk13: i32,
46    unk14: i32,
47    unk15: i32,
48    unk16: i32,
49    unk17: i32,
50    /// The original allocation returned by `sceMeMalloc`.
51    ///
52    /// This may differ from `edram_addr` because `edram_addr` is aligned forward to a 64-byte
53    /// boundary. The original pointer must be retained so it can be passed to `sceMeFree`.
54    alloc_mem: *mut u8,
55}

57/// The possible codec kinds.
58#[repr(u16)]
59#[derive(Debug, Clone, Copy, PartialEq, PartialOrd, Eq, Ord, Hash)]
60pub enum CodecKind {
61    Atrac3Plus = 0x1000,
62    Atrac3 = 0x1001,
63    Mp3    = 0x1002,
64    Aac    = 0x1003,
65}
