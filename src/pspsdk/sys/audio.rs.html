77#[psp_stub(libname = "sceAudio", flags = 0x4009, use_crate)]
78unsafe extern "C" {
79    /// Allocate and initialize a hardware output channel.
80    ///
81    /// # Parameters
82    ///
83    /// - `channel`: The channel ID to reserve. Pass [`AudioChannelId::NEXT`] to get the first
84    ///   available channel.
85    /// - `sample_count`: The number of samples that can be output on the channel per output call.
86    ///   It must be a value between [`AUDIO_SAMPLE_MIN`] and [`AUDIO_SAMPLE_MAX`], and it must be
87    ///   aligned to 64 bytes. Use [`audio_sample_align`] to align it.
88    /// - `format`: The output format to use for the channel.
89    ///
90    /// # Return value
91    ///
92    /// The channel ID on success, an error value otherwise.
93    #[nid(0x5EC81C55)]
94    pub safe fn sceAudioChReserve(
95        channel: AudioChannelId, sample_count: i32, format: AudioFormats,
96    ) -> SceResult<AudioChannelId>;
97}
98
99/// Make the given sample count a multiple of 64.
100pub const fn audio_sample_align(sample_count: i32) -> i32 {
101    (sample_count + 63) & !63
102}

104impl AudioChannelId {
105    /// Channel 0.
106    pub const CHANNEL_0: AudioChannelId = unsafe { Self::from_raw_unchecked(0) };
107    /// Channel 1.
108    pub const CHANNEL_1: AudioChannelId = unsafe { Self::from_raw_unchecked(1) };
109    /// Channel 2.
110    pub const CHANNEL_2: AudioChannelId = unsafe { Self::from_raw_unchecked(2) };
111    /// Channel 3.
112    pub const CHANNEL_3: AudioChannelId = unsafe { Self::from_raw_unchecked(3) };
113    /// Channel 4.
114    pub const CHANNEL_4: AudioChannelId = unsafe { Self::from_raw_unchecked(4) };
115    /// Channel 5.
116    pub const CHANNEL_5: AudioChannelId = unsafe { Self::from_raw_unchecked(5) };
117    /// Channel 6.
118    pub const CHANNEL_6: AudioChannelId = unsafe { Self::from_raw_unchecked(6) };
119    /// Channel 7.
120    pub const CHANNEL_7: AudioChannelId = unsafe { Self::from_raw_unchecked(7) };
121    /// Channel value to request a the next channel on functions like [`sceAudioChReserve`].
122    pub const NEXT: AudioChannelId = unsafe { Self::from_raw_unchecked(0xFFFFFFFF) };
123
124    /// Create a new channel number from a raw value.
125    ///
126    /// This functions checks for the value of `raw` to be a in the range of possible `SceChannel`
127    /// values used by the PSP OS, returning an [`None`] otherwise.
128    pub const fn from_raw(raw: u32) -> Option<Self> {
129        match raw {
130            0u32..=7 => Some(unsafe { Self::from_raw_unchecked(raw) }),
131            0xFFFFFFFF => Some(Self::NEXT),
132            _ => None,
133        }
134    }
135
136    /// Create a new channel number structure from a raw value without checking value range.
137    ///
138    /// # Safety
139    ///
140    /// Immediate language UB if `val` is not within the valid range for this
141    /// type, as it violates the validity invariant.
142    #[inline]
143    pub const unsafe fn from_raw_unchecked(raw: u32) -> Self {
144        Self(raw)
145    }
146
147    #[inline]
148    pub const fn to_inner(self) -> u32 {
149        // SAFETY: pattern types are always legal values of their base type
150        // (Not using `.0` because that has perf regressions.)
151        unsafe { core::mem::transmute(self) }
152    }
153}

160            0..=7 => Some(unsafe { Self::from_raw_unchecked(ok_value) }),
161            _ => None,
162        }
163    }
