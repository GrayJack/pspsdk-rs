26impl Thread {
27    #[allow(private_interfaces, clippy::missing_safety_doc, reason = "Internal API")]
28    pub unsafe fn with_attr(
29        stack: usize, attr: ThreadAttributes, init: Box<ThreadInit>,
30    ) -> io::Result<Thread> {
31        extern "C" fn thread_start(args: SceSize, argp: *mut c_void) -> SceResult<u32> {
32            if argp.is_null() || args != size_of::<*mut ()>() {
33                return SceResult::new(u32::MAX);
34            }
35
36            // SAFETY: we are simply recreating the box that was leaked earlier.
37            let init_ptr = unsafe { *(argp as *mut *mut ThreadInit) };
38            let init: Box<ThreadInit> = unsafe { Box::from_raw(init_ptr) };
39            let rust_start = init.init();
40            rust_start();
41
42            unsafe { crate::sys::thread_local::key::run_dtors() };
43
44            SceResult::new(0)
45        }
46
47        let thread_name = init.handle.cname().unwrap_or(c"");
48
49        let thread = unsafe {
50            sceKernelCreateThread(
51                thread_name.as_ptr().cast(),
52                thread_start,
53                0x20,
54                stack,
55                attr,
56                None,
57            )
58            .map_err(Into::<io::Error>::into)?
59        };
60
61        let mut init_ptr = Box::into_raw(init);
62        let ptr = &raw mut init_ptr;
63        let res = unsafe { sceKernelStartThread(thread, size_of_val(&init_ptr), ptr.cast()) };
64
65        if let Some(err) = res.err() {
66            // The thread failed to start and as a result data was not consumed. Therefore, it is
67            // safe to reconstruct the box so that it gets deallocated.
68            drop(unsafe { Box::from_raw(init_ptr) });
69            let _ = unsafe { sceKernelDeleteThread(thread) };
70            Err(err.into())
71        } else {
72            Ok(Thread { id: thread })
73        }
74    }
75
76    #[allow(private_interfaces, clippy::missing_safety_doc, reason = "Internal API")]
77    pub unsafe fn new(stack: usize, init: Box<ThreadInit>) -> io::Result<Thread> {
78        let attr = cfg_select! {
79            feature = "kernel" => ThreadAttributes::default(),
80            all(prx, not(feature = "kernel")) => ThreadAttributes::UserMode,
81            pbp => ThreadAttributes::UserMode | ThreadAttributes::UseVFPU,
82            _ => ThreadAttributes::default(),
83        };
84        unsafe { Thread::with_attr(stack, attr, init) }
85    }

87    pub fn join(self) {
88        let id = self.into_id();
89
90        let ret = sceKernelWaitThreadEndCB(id, None);
91        assert!(ret.is_ok(), "failed to join thread: {}", unsafe {
92            SceError::from_raw_unchecked(ret.as_inner())
93        });
94
95        let ret = unsafe { sceKernelDeleteThread(id) };
96        assert!(ret.is_ok(), "failed to delete join thread: {}", unsafe {
97            SceError::from_raw_unchecked(ret.as_inner())
98        });
99    }

134pub fn sleep(dur: Duration) {
135    let mut micros =
136        dur.as_micros() + if !dur.subsec_nanos().is_multiple_of(1_000) { 1 } else { 0 };
137
138    while micros > 0 {
139        let chunk = micros.min(u32::MAX as u128) as u32;
140        let res = sceKernelDelayThreadCB(chunk);
141
142        if res.is_ok() {
143            micros -= chunk as u128;
144        }
145    }
146}

148pub fn sleep_until(deadline: Instant) {
149    // The clock source used for `sleep` might not be the same used for `Instant`.
150    // Since this function *must not* return before the deadline, we recheck the
151    // time after every call to `sleep`. See #149935 for an example of this
152    // occurring on older Windows systems.
153    while let Some(delay) = deadline.checked_duration_since(Instant::now()) {
154        // Sleep for the estimated time remaining until the deadline.
155        //
156        // If your system has a better way of estimating the delay time or
157        // provides a way to sleep until an absolute time, specialize this
158        // function for your system.
159        sleep(delay);
160    }
