364/// Possible font pixel formats.
365#[repr(u32)]
366#[derive(Debug, Clone, Copy, PartialEq, PartialOrd, Eq, Ord, Hash, Default)]
367pub enum PixelFormat {
368    /// This formats as 2 pixels in one byte (8 bits).
369    ///
370    /// The higher half is to be placed at the left side of the screen. The lower half is to be
371    /// placed at the right side of the screen.
372    ///
373    /// `0x0` means minimal brightness, and `0xF` means maximum brightness.
374    #[default]
375    L4    = 0,
376    /// Probably the oposite of `L4`, but not used.
377    R4    = 1,
378    /// This formats as 1 pixel in one byte (8 bits).
379    ///
380    /// `0x00` means minimal brightness, and `0xFF` means maximum brightness.
381    Byte  = 2,
382    /// Probably 24bit pixels, not used.
383    Bit24 = 3,
384    /// Probably 32bit pixels, not used.
385    Bit32 = 4,
386}
