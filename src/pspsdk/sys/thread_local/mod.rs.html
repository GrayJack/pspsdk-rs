10/// Run a callback in a scenario which must not unwind (such as a `extern "C"
11/// fn` declared in a user crate). If the callback unwinds anyway, then
12/// `rtabort` with a message about thread local panicking on drop.
13#[inline]
14#[allow(dead_code)]
15fn abort_on_dtor_unwind(f: impl FnOnce()) {
16    // Using a guard like this is lower cost.
17    let guard = DtorUnwindGuard;
18    f();
19    core::mem::forget(guard);
20
21    struct DtorUnwindGuard;
22    impl Drop for DtorUnwindGuard {
23        #[inline]
24        fn drop(&mut self) {
25            // This is not terribly descriptive, but it doesn't need to be as we'll
26            // already have printed a panic message at this point.
27            crate::rtabort!("thread local panicked on drop");
28        }
29    }
30}
