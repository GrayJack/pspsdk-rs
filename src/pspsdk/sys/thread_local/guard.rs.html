1use core::ptr;
2
3use crate::sys::thread_local::key::{set, LazyKey};
4
5/// On platforms with key-based TLS, the system runs the destructors for us.
6/// We still have to make sure that [`crate::rt::thread_cleanup`] is called,
7/// however. This is done by deferring the execution of a TLS destructor to
8/// the next round of destruction inside the TLS destructors.
9pub fn enable() {
10    const DEFER: *mut u8 = ptr::without_provenance_mut(1);
11    const RUN: *mut u8 = ptr::without_provenance_mut(2);
12
13    static CLEANUP: LazyKey = LazyKey::new(Some(run));
14
15    unsafe { set(CLEANUP.force(), DEFER) }
16
17    unsafe extern "C" fn run(state: *mut u8) {
18        if state == DEFER {
19            // Make sure that this function is run again in the next round of
20            // TLS destruction. If there is no further round, there will be leaks,
21            // but that's okay, `thread_cleanup` is not guaranteed to be called.
22            unsafe { set(CLEANUP.force(), RUN) }
23        } else {
24            debug_assert_eq!(state, RUN);
25            // If the state is still RUN in the next round of TLS destruction,
26            // it means that no other TLS destructors defined by this runtime
27            // have been run, as they would have set the state to DEFER.
28            crate::rt::thread_cleanup();
29        }
30    }
31}
