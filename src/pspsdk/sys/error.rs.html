70impl SceError {
71    /// Create a new Error structure from a raw value.
72    ///
73    /// This functions checks for the value of `raw` to be a in the range of possible SceError
74    /// values used by the PSP OS and homebrews, returning an [`None`] otherwise. That means
75    /// that this function allows the creation of custom error codes outside of the range used in by
76    /// the PSP OS.
77    ///
78    /// # Note
79    /// This type has as constants all the OS error codes. We encourage to use them when possible.
80    pub const fn from_raw(raw: u32) -> Option<Self> {
81        if let 0x80000001..=0xFFFFFFFF = raw {
82            Some(unsafe { Self::from_raw_unchecked(raw) })
83        } else {
84            None
85        }
86    }
87
88    /// Create a new Error structure from a raw value without checking value range.
89    ///
90    /// # Safety
91    ///
92    /// Immediate language UB if `val` is not within the valid range for this
93    /// type, as it violates the validity invariant.
94    #[inline]
95    pub const unsafe fn from_raw_unchecked(raw: u32) -> Self {
96        // SAFETY: Caller promised that `val` is within the valid range.
97        unsafe { core::mem::transmute(raw) }
98    }
99
100    #[inline]
101    pub const fn to_inner(self) -> u32 {
102        // SAFETY: pattern types are always legal values of their base type
103        // (Not using `.0` because that has perf regressions.)
104        unsafe { core::mem::transmute(self) }
105    }
