16const MAX_ARGC: usize = 19;
17
18/// Process `argc_bytes` and `argp` from `module_start` and creates a `argc` and `argv` to pass to
19/// be passed to be used later. when starting the module main thread.
20///
21/// # Safety
22/// `argp` must be valid for `argc_bytes`. And this function is only to be used with `module_start`.
23pub unsafe fn process_argc_argv(
24    argc_bytes: usize, argp: *mut u8,
25) -> (usize, [*mut u8; MAX_ARGC + 1]) {
26    let mut argv: [*mut u8; 20] = [core::ptr::null_mut(); 20];
27    let mut argc = 0;
28    let mut loc = 0;
29    let ptr: *mut u8 = argp.cast();
30
31    if !ptr.is_null() {
32        while loc < argc_bytes {
33            unsafe {
34                argv[argc] = ptr.add(loc);
35
36                let arg_len = crate::private::strlen(argv[argc].cast()) + 1;
37
38                loc += arg_len;
39                argc += 1;
40
41                if argc == 19 {
42                    break;
43                }
44            }
45        }
46    }
47
48    (argc, argv)
49}

51/// Initialization of stuff required to live on `module_start`. Returns `argc` and `argv` tuple.
52///
53/// For barebones projects, it is the user responsibility to call this function on `module_start`.
54///
55/// # Safety
56/// `argp` must be valid for `argc_bytes`. And this function is only to be used with `module_start`.
57pub unsafe fn module_start_init(
58    argc_bytes: usize, argp: *mut core::ffi::c_void,
59) -> (usize, [*mut u8; MAX_ARGC + 1]) {
60    // Set OS functions for pspsdk::io module
61    crate::set_psp_os_functions();
62
63    unsafe { process_argc_argv(argc_bytes, argp.cast()) }
64}

66/// Initialize the current working directory.
67///
68/// # Safety
69/// `arg0` must be valid for null-terminated string.
70pub unsafe fn init_cwd(arg0: *mut u8) {
71    if arg0.is_null() {
72        return;
73    }
74
75    unsafe {
76        let mut len = 0;
77        while *arg0.add(len) != 0 {
78            len += 1;
79        }
80
81        // Truncate until last '/'
82        while len > 0 && *arg0.add(len - 1) != b'/' {
83            len -= 1;
84        }
85
86        if len > 0 {
87            let tmp = *arg0.add(len);
88            *arg0.add(len) = 0;
89            let _res = crate::sys::io::sceIoChdir(arg0 as *const u8);
90            *arg0.add(len) = tmp;
91        }
92    }
93}

95pub(crate) static MAIN_THREAD_ID: SyncUnsafeCell<Option<ThreadId>> = SyncUnsafeCell::new(None);
96pub(crate) static CUSTOM_CLEANUP: SyncUnsafeCell<Option<fn()>> = SyncUnsafeCell::new(None);

98pub(crate) fn main_thread_id() -> Option<ThreadId> {
99    let id = unsafe { &*MAIN_THREAD_ID.get() };
100    *id
101}

103pub(crate) fn custom_cleanup() -> Option<fn()> {
104    let res = unsafe { &*CUSTOM_CLEANUP.get() };
105    *res
106}

108/// Sets a custom cleanup procedure that will be called on [`exit`] and on the set callback by
109/// [`enable_home_button`].
110///
111/// It returns the previously set value.
112///
113/// [`exit`]: crate::process::exit
114/// [`enable_home_button`]: crate::enable_home_button
115pub fn set_custom_cleanup(f: fn()) -> Option<fn()> {
116    let cleanup = CUSTOM_CLEANUP.get();
117    let previous = unsafe { cleanup.read() };
118    unsafe {
119        cleanup.write(Some(f));
120    }
121
122    previous
123}
124pub(crate) unsafe fn init(_argc: usize, argv: *const *mut u8) {
125    // Ideally this is called on module_start, but people doing barebones project may forget, so we
126    // call it here too.
127    crate::set_psp_os_functions();
128
129    let main_thread_id = sys::thread::sceKernelGetThreadId().into_result();
130
131    if let Ok(id) = main_thread_id {
132        unsafe { MAIN_THREAD_ID.get().write_volatile(Some(id)) };
133    }
134
135    unsafe {
136        init_cwd(*argv);
137    }
138
139    // Enable
140    if cfg!(pbp) {
141        crate::enable_home_button();
142    }
143}

145/// Cleanup procedure to be run after `main`
146///
147/// If you are a plugin, remember to do this manually
148pub(crate) fn cleanup() {
149    use crate::sync::nonpoison::Once;
150
151    static CLEANUP: Once = Once::new();
152    CLEANUP.call_once(|| unsafe {
153        if let Some(f) = custom_cleanup() {
154            f();
155        }
156
157        // Flush stdout and disable buffering.
158        crate::io::cleanup();
159        // SAFETY: Only called once during runtime cleanup.
160        sys::cleanup();
161    });
162}

164fn handle_rt_panic<T>(e: alloc::boxed::Box<dyn core::any::Any + Send>) -> T {
165    core::mem::forget(e);
166    crate::rtabort!("initialization or cleanup bug")
167}

169// To reduce the generated code of the new `psp_start`, this function is doing
170// the real work.
171#[cfg(not(test))]
172fn psp_start_internal(
173    main: &(dyn Fn() -> i32 + Sync + core::panic::RefUnwindSafe), argc: usize, argv: *const *mut u8,
174) -> isize {
175    // Guard against the code called by this function from unwinding outside of the Rust-controlled
176    // code, which is UB. This is a requirement imposed by a combination of how the
177    // `#[lang="start"]` attribute is implemented as well as by the implementation of the panicking
178    // mechanism itself.
179    //
180    // There are a couple of instances where unwinding can begin. First is inside of the
181    // `rt::init`, `rt::cleanup` and similar functions controlled by std. In those instances a
182    // panic is a std implementation bug. A quite likely one too, as there isn't any way to
183    // prevent std from accidentally introducing a panic to these functions. Another is from
184    // user code from `main` or, more nefariously, as described in e.g. issue #86030.
185    //
186    // We use `catch_unwind` with `handle_rt_panic` instead of `abort_unwind` to make the error in
187    // case of a panic a bit nicer.
188    panicking::catch_unwind(move || {
189        // SAFETY: Only called once during runtime initialization.
190        unsafe { init(argc, argv) };
191
192        let ret_code = panicking::catch_unwind(main).unwrap_or_else(move |payload| {
193            use crate::sys::SceError;
194
195            // Carefully dispose of the panic payload.
196            let payload = panicking::AssertUnwindSafe(payload);
197            panicking::catch_unwind(move || drop({ payload }.0)).unwrap_or_else(move |e| {
198                core::mem::forget(e); // do *not* drop the 2nd payload
199                crate::rtabort!("drop of the panic payload panicked");
200            });
201            // Return error code for panicking programs.
202            SceError::NOT_INIT.to_inner().cast_signed()
203        });
204        let ret_code = ret_code as isize;
205
206        cleanup();
207
208        // Guard against multiple threads calling `libc::exit` concurrently.
209        // See the documentation for `unique_thread_exit` for more information.
210        // crate::sys::exit::unique_thread_exit();
211
212        ret_code
213    })
214    .unwrap_or_else(handle_rt_panic)
215}

217// WARNING: `argc_bytes` and `argp` are not the same as `argc` and `argv`.
218//
219// The PSP system allows to pass arbitrary data to the `sceKernelCreateThread` function pointer.
220// It can be a struct reference, and array of data, etc.
221//
222// It just happen that for module start the data passed is passed as a form similar to `argc` and
223// `argv`, but it needs to be processed before.
224//
225// In this regard, this is different from `rust_start`
226#[cfg(not(any(test, doctest)))]
227#[doc(hidden)]
228pub fn psp_start<T: crate::process::Termination + 'static>(
229    main: fn() -> T, argc: usize, argv: *const *mut u8,
230) -> isize {
231    psp_start_internal(&move || main().report().to_i32(), argc, argv)
232}
