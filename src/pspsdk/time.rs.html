5//! There are multiple ways to create a new [`Duration`]:
6//!
7//! ```
8//! # use pspsdk::time::Duration;
9//! let five_seconds = Duration::from_secs(5);
10//! assert_eq!(five_seconds, Duration::from_millis(5_000));

13//!
14//! let ten_seconds = Duration::from_secs(10);
15//! let seven_nanos = Duration::from_nanos(7);

47/// A measurement of a monotonically nondecreasing clock.
48/// Opaque and useful only with [`Duration`].
49///
50/// Instants are always guaranteed, barring [platform bugs], to be no less than any previously
51/// measured instant when created, and are often useful for tasks such as measuring
52/// benchmarks or timing how long an operation takes.
53///
54/// Note, however, that instants are **not** guaranteed to be **steady**. In other
55/// words, each tick of the underlying clock might not be the same length (e.g.
56/// some seconds may be longer than others). An instant may jump forwards or
57/// experience time dilation (slow down or speed up), but it will never go
58/// backwards.
59/// As part of this non-guarantee it is also not specified whether system suspends count as
60/// elapsed time or not. The behavior varies across platforms and Rust versions.
61///
62/// Instants are opaque types that can only be compared to one another. There is
63/// no method to get "the number of seconds" from an instant. Instead, it only
64/// allows measuring the duration between two instants (or comparing two
65/// instants).
66///
67/// The size of an `Instant` struct may vary depending on the target operating
68/// system.
69///
70/// Example:
71///
72/// ```no_run
73/// use std::thread::sleep;
74///
75/// use pspsdk::time::{Duration, Instant};
76///
77/// fn main() {
78///     let now = Instant::now();
79///
80///     // we sleep for 2 seconds
81///     sleep(Duration::new(2, 0));
82///     // it prints '2'
83///     println!("{}", now.elapsed().as_secs());
84/// }
85/// ```
86///
87/// [platform bugs]: Instant#monotonicity
88///
89/// # OS-specific behaviors
90///
91/// An `Instant` is a wrapper around system-specific types and it may behave
92/// differently depending on the underlying operating system. For example,
93/// the following snippet is fine on Linux but panics on macOS:
94///
95/// ```no_run
96/// use pspsdk::time::{Duration, Instant};
97///
98/// let now = Instant::now();
99/// let max_seconds = u64::MAX / 1_000_000_000;
100/// let duration = Duration::new(max_seconds, 0);
101/// println!("{:?}", now + duration);
102/// ```

131/// ## Monotonicity
132///
133/// On all platforms `Instant` will try to use an OS API that guarantees monotonic behavior
134/// if available, which is the case for all [tier 1] platforms.
135/// In practice such guarantees are – under rare circumstances – broken by hardware, virtualization
136/// or operating system bugs. To work around these bugs and platforms not offering monotonic clocks
137/// [`duration_since`], [`elapsed`] and [`sub`] saturate to zero. In older Rust versions this
138/// lead to a panic instead. [`checked_duration_since`] can be used to detect and handle situations
139/// where monotonicity is violated, or `Instant`s are subtracted in the wrong order.
140///
141/// This workaround obscures programming errors where earlier and later instants are accidentally
142/// swapped. For this reason future Rust versions may reintroduce panics.
143///
144/// [tier 1]: https://doc.rust-lang.org/rustc/platform-support.html
145/// [`duration_since`]: Instant::duration_since
146/// [`elapsed`]: Instant::elapsed
147/// [`sub`]: Instant::sub
148/// [`checked_duration_since`]: Instant::checked_duration_since
149#[derive(Copy, Clone, PartialEq, Eq, PartialOrd, Ord, Hash)]
150pub struct Instant(time::Instant);

152/// A measurement of the system clock, useful for talking to
153/// external entities like the file system or other processes.
154///
155/// Distinct from the [`Instant`] type, this time measurement **is not
156/// monotonic**. This means that you can save a file to the file system, then
157/// save another file to the file system, **and the second file has a
158/// `SystemTime` measurement earlier than the first**. In other words, an
159/// operation that happens after another operation in real time may have an
160/// earlier `SystemTime`!
161///
162/// Consequently, comparing two `SystemTime` instances to learn about the
163/// duration between them returns a [`Result`] instead of an infallible [`Duration`]
164/// to indicate that this sort of time drift may happen and needs to be handled.
165///
166/// Although a `SystemTime` cannot be directly inspected, the [`UNIX_EPOCH`]
167/// constant is provided in this module as an anchor in time to learn
168/// information about a `SystemTime`. By calculating the duration from this
169/// fixed point in time, a `SystemTime` can be converted to a human-readable time,
170/// or perhaps some other string representation.
171///
172/// The size of a `SystemTime` struct may vary depending on the target operating
173/// system.
174///
175/// A `SystemTime` does not count leap seconds.
176/// `SystemTime::now()`'s behaviour around a leap second
177/// is the same as the operating system's wall clock.
178/// The precise behaviour near a leap second
179/// (e.g. whether the clock appears to run slow or fast, or stop, or jump)
180/// depends on platform and configuration,
181/// so should not be relied on.
182///
183/// Example:
184///
185/// ```no_run
186/// use std::thread::sleep;
187///
188/// use pspsdk::time::{Duration, Instant};
189///
190/// fn main() {
191///     let now = SystemTime::now();
192///
193///     // we sleep for 2 seconds
194///     sleep(Duration::new(2, 0));
195///     match now.elapsed() {
196///         Ok(elapsed) => {
197///             // it prints '2'
198///             println!("{}", elapsed.as_secs());
199///         },
200///         Err(e) => {
201///             // an error occurred!
202///             println!("Error: {e:?}");
203///         },
204///     }
205/// }
206/// ```
207// # Platform-specific behavior

209// The precision of `SystemTime` can depend on the underlying OS-specific time format.
210// For example, on Windows the time is represented in 100 nanosecond intervals whereas Linux
211// can represent nanosecond intervals.
212//
213// The following system calls are [currently] being used by `now()` to find out
214// the current time:

216// |  Platform |               System call                                            |

222// | SOLID     | `SOLID_RTC_ReadTime`                                                 |
223// | WASI      | [__wasi_clock_time_get (Realtime Clock)]                             |

308    /// Returns the amount of time elapsed from another instant to this one,
309    /// or None if that instant is later than this one.
310    ///
311    /// Due to [monotonicity bugs], even under correct logical ordering of the passed `Instant`s,
312    /// this method can return `None`.
313    ///
314    /// [monotonicity bugs]: Instant#monotonicity
315    ///
316    /// # Examples
317    ///
318    /// ```no_run
319    /// use std::thread::sleep;
320    ///

327    /// println!("{:?}", now.checked_duration_since(new_now)); // None
328    /// ```
329    #[must_use]
330    pub fn checked_duration_since(&self, earlier: Instant) -> Option<Duration> {
331        self.0.checked_sub_instant(&earlier.0)
332    }

335    /// or zero duration if that instant is later than this one.

355    /// Returns the amount of time elapsed since this instant.
356    ///
357    /// # Panics
358    ///
359    /// Previous Rust versions panicked when the current time was earlier than self. Currently this
360    /// method returns a Duration of zero in that case. Future versions may reintroduce the panic.
361    /// See [Monotonicity].
362    ///
363    /// [Monotonicity]: Instant#monotonicity
364    ///
365    /// # Examples
366    ///
367    /// ```no_run
368    /// use std::thread::sleep;
369    ///
370    /// use pspsdk::time::{Duration, Instant};
371    ///
372    /// let instant = Instant::now();
373    /// let three_secs = Duration::from_secs(3);
374    /// sleep(three_secs);
375    /// assert!(instant.elapsed() >= three_secs);
376    /// ```
377    #[must_use]
378    pub fn elapsed(&self) -> Duration {
379        Instant::now() - *self
380    }

382    /// Returns `Some(t)` where `t` is the time `self + duration` if `t` can be represented as
383    /// `Instant` (which means it's inside the bounds of the underlying data structure), `None`
384    /// otherwise.
385    pub fn checked_add(&self, duration: Duration) -> Option<Instant> {
386        self.0.checked_add_duration(&duration).map(Instant)
387    }

389    /// Returns `Some(t)` where `t` is the time `self - duration` if `t` can be represented as
390    /// `Instant` (which means it's inside the bounds of the underlying data structure), `None`
391    /// otherwise.
392    pub fn checked_sub(&self, duration: Duration) -> Option<Instant> {
393        self.0.checked_sub_duration(&duration).map(Instant)
394    }

454impl SystemTime {
455    /// Represents the maximum value representable by [`SystemTime`] on this platform.
456    ///
457    /// This value differs a lot between platforms, but it is always the case
458    /// that any positive addition of a [`Duration`], whose value is greater
459    /// than or equal to the time precision of the operating system, to
460    /// [`SystemTime::MAX`] will fail.
461    ///
462    /// # Examples
463    ///
464    /// ```no_run
465    /// use pspsdk::time::{Duration, SystemTime};
466    ///
467    /// // Adding zero will change nothing.
468    /// assert_eq!(SystemTime::MAX.checked_add(Duration::ZERO), Some(SystemTime::MAX));
469    ///
470    /// // But adding just one second will already fail ...
471    /// //
472    /// // Keep in mind that this in fact may succeed, if the Duration is
473    /// // smaller than the time precision of the operating system, which
474    /// // happens to be 1ns on most operating systems, with Windows being the
475    /// // notable exception by using 100ns, hence why this example uses 1s.
476    /// assert_eq!(SystemTime::MAX.checked_add(Duration::new(1, 0)), None);
477    ///
478    /// // Utilize this for saturating arithmetic to improve error handling.
479    /// // In this case, we will use a certificate with a timestamp in the
480    /// // future as a practical example.
481    /// let configured_offset = Duration::from_secs(60 * 60 * 24);
482    /// let valid_after = SystemTime::now().checked_add(configured_offset).unwrap_or(SystemTime::MAX);
483    /// ```
484    pub const MAX: SystemTime = SystemTime(time::SystemTime::MAX);
485    /// Represents the minimum value representable by [`SystemTime`] on this platform.
486    ///
487    /// This value differs a lot between platforms, but it is always the case
488    /// that any positive subtraction of a [`Duration`] from, whose value is
489    /// greater than or equal to the time precision of the operating system, to
490    /// [`SystemTime::MIN`] will fail.
491    ///
492    /// Depending on the platform, this may be either less than or equal to
493    /// [`SystemTime::UNIX_EPOCH`], depending on whether the operating system
494    /// supports the representation of timestamps before the Unix epoch or not.
495    /// However, it is always guaranteed that a [`SystemTime::UNIX_EPOCH`] fits
496    /// between a [`SystemTime::MIN`] and [`SystemTime::MAX`].
497    ///
498    /// # Examples
499    ///
500    /// ```
501    /// use pspsdk::time::{Duration, SystemTime};
502    ///
503    /// // Subtracting zero will change nothing.
504    /// assert_eq!(SystemTime::MIN.checked_sub(Duration::ZERO), Some(SystemTime::MIN));
505    ///
506    /// // But subtracting just one second will already fail.
507    /// //
508    /// // Keep in mind that this in fact may succeed, if the Duration is
509    /// // smaller than the time precision of the operating system, which
510    /// // happens to be 1ns on most operating systems, with Windows being the
511    /// // notable exception by using 100ns, hence why this example uses 1s.
512    /// assert_eq!(SystemTime::MIN.checked_sub(Duration::new(1, 0)), None);
513    ///
514    /// // Utilize this for saturating arithmetic to improve error handling.
515    /// // In this case, we will use a cache expiry as a practical example.
516    /// let configured_expiry = Duration::from_secs(60 * 3);
517    /// let expiry_threshold =
518    ///     SystemTime::now().checked_sub(configured_expiry).unwrap_or(SystemTime::MIN);
519    /// ```
520    pub const MIN: SystemTime = SystemTime(time::SystemTime::MIN);
521    /// An anchor in time which can be used to create new `SystemTime` instances or
522    /// learn about where in time a `SystemTime` lies.
523    // NOTE! this documentation is duplicated, here and in std::time::UNIX_EPOCH.
524    // The two copies are not quite identical, because of the difference in naming.
525    /// This constant is defined to be "1970-01-01 00:00:00 UTC" on all systems with
526    /// respect to the system clock. Using `duration_since` on an existing
527    /// `SystemTime` instance can tell how far away from this point in time a
528    /// measurement lies, and using `UNIX_EPOCH + duration` can be used to create a
529    /// `SystemTime` instance to represent another fixed point in time.
530    ///
531    /// `duration_since(UNIX_EPOCH).unwrap().as_secs()` returns
532    /// the number of non-leap seconds since the start of 1970 UTC.
533    /// This is a POSIX `time_t` (as a `u64`),
534    /// and is the same time representation as used in many Internet protocols.
535    ///
536    /// # Examples
537    ///
538    /// ```no_run
539    /// use pspsdk::time::SystemTime;
540    ///
541    /// match SystemTime::now().duration_since(SystemTime::UNIX_EPOCH) {
542    ///     Ok(n) => pspsdk::println!("1970-01-01 00:00:00 UTC was {} seconds ago!", n.as_secs()),
543    ///     Err(_) => panic!("SystemTime before UNIX EPOCH!"),
544    /// }
545    /// ```
546    pub const UNIX_EPOCH: SystemTime = UNIX_EPOCH;
547
548    /// Returns the system time corresponding to "now".
549    ///
550    /// # Examples
551    ///
552    /// ```
553    /// use pspsdk::time::SystemTime;
554    ///
555    /// let sys_time = SystemTime::now();
556    /// ```
557    #[must_use]
558    pub fn now() -> SystemTime {
559        SystemTime(time::SystemTime::now())
560    }

562    /// Returns the amount of time elapsed from an earlier point in time.
563    ///
564    /// This function may fail because measurements taken earlier are not
565    /// guaranteed to always be before later measurements (due to anomalies such
566    /// as the system clock being adjusted either forwards or backwards).
567    /// [`Instant`] can be used to measure elapsed time without this risk of failure.
568    ///
569    /// If successful, <code>[Ok]\([Duration])</code> is returned where the duration represents
570    /// the amount of time elapsed from the specified measurement to this one.
571    ///
572    /// Returns an [`Err`] if `earlier` is later than `self`, and the error
573    /// contains how far from `self` the time is.
574    ///
575    /// # Examples
576    ///
577    /// ```no_run
578    /// use pspsdk::time::SystemTime;
579    ///
580    /// let sys_time = SystemTime::now();
581    /// let new_sys_time = SystemTime::now();
582    /// let difference = new_sys_time
583    ///     .duration_since(sys_time)
584    ///     .expect("Clock may have gone backwards");
585    /// pspsdk::println!("{difference:?}");
586    /// ```
587    pub fn duration_since(&self, earlier: SystemTime) -> Result<Duration, SystemTimeError> {
588        self.0.sub_time(&earlier.0).map_err(SystemTimeError)
589    }

591    /// Returns the difference from this system time to the
592    /// current clock time.
593    ///
594    /// This function may fail as the underlying system clock is susceptible to
595    /// drift and updates (e.g., the system clock could go backwards), so this
596    /// function might not always succeed. If successful, <code>[Ok]\([Duration])</code> is
597    /// returned where the duration represents the amount of time elapsed from
598    /// this time measurement to the current time.
599    ///
600    /// To measure elapsed time reliably, use [`Instant`] instead.
601    ///
602    /// Returns an [`Err`] if `self` is later than the current system time, and
603    /// the error contains how far from the current system time `self` is.
604    ///
605    /// # Examples
606    ///
607    /// ```no_run
608    /// use std::thread::sleep;
609    ///
610    /// use pspsdk::time::{Duration, SystemTime};
611    ///
612    /// let sys_time = SystemTime::now();
613    /// let one_sec = Duration::from_secs(1);
614    /// sleep(one_sec);
615    /// assert!(sys_time.elapsed().unwrap() >= one_sec);
616    /// ```
617    pub fn elapsed(&self) -> Result<Duration, SystemTimeError> {
618        SystemTime::now().duration_since(*self)
619    }
620
621    /// Returns `Some(t)` where `t` is the time `self + duration` if `t` can be represented as
622    /// `SystemTime` (which means it's inside the bounds of the underlying data structure), `None`
623    /// otherwise.
624    ///
625    /// In the case that the `duration` is smaller than the time precision of the operating
626    /// system, `Some(self)` will be returned.
627    pub fn checked_add(&self, duration: Duration) -> Option<SystemTime> {
628        self.0.checked_add_duration(&duration).map(SystemTime)
629    }
630
631    /// Returns `Some(t)` where `t` is the time `self - duration` if `t` can be represented as
632    /// `SystemTime` (which means it's inside the bounds of the underlying data structure), `None`
633    /// otherwise.
634    ///
635    /// In the case that the `duration` is smaller than the time precision of the operating
636    /// system, `Some(self)` will be returned.
637    pub fn checked_sub(&self, duration: Duration) -> Option<SystemTime> {
638        self.0.checked_sub_duration(&duration).map(SystemTime)
639    }
640
641    /// Saturating [`SystemTime`] addition, computing `self + duration`,
642    /// returning [`SystemTime::MAX`] if overflow occurred.
643    ///
644    /// In the case that the `duration` is smaller than the time precision of
645    /// the operating system, `self` will be returned.
646    pub fn saturating_add(&self, duration: Duration) -> SystemTime {
647        self.checked_add(duration).unwrap_or(SystemTime::MAX)
648    }
649
650    /// Saturating [`SystemTime`] subtraction, computing `self - duration`,
651    /// returning [`SystemTime::MIN`] if overflow occurred.
652    ///
653    /// In the case that the `duration` is smaller than the time precision of
654    /// the operating system, `self` will be returned.
655    pub fn saturating_sub(&self, duration: Duration) -> SystemTime {
656        self.checked_sub(duration).unwrap_or(SystemTime::MIN)
657    }
658
659    /// Saturating computation of time elapsed from an earlier point in time,
660    /// returning [`Duration::ZERO`] in the case that `earlier` is later or
661    /// equal to `self`.
662    ///
663    /// # Examples
664    ///
665    /// ```no_run
666    /// use pspsdk::time::{Duration, SystemTime};
667    ///
668    /// let now = SystemTime::now();
669    /// let prev = now.saturating_sub(Duration::new(1, 0));
670    ///
671    /// // now - prev should return non-zero.
672    /// assert_eq!(now.saturating_duration_since(prev), Duration::new(1, 0));
673    /// assert!(now.duration_since(prev).is_ok());
674    ///
675    /// // prev - now should return zero (and fail with the non-saturating).
676    /// assert_eq!(prev.saturating_duration_since(now), Duration::ZERO);
677    /// assert!(prev.duration_since(now).is_err());
678    ///
679    /// // now - now should return zero (and work with the non-saturating).
680    /// assert_eq!(now.saturating_duration_since(now), Duration::ZERO);
681    /// assert!(now.duration_since(now).is_ok());
682    /// ```
683    pub fn saturating_duration_since(&self, earlier: SystemTime) -> Duration {
684        self.duration_since(earlier).unwrap_or(Duration::ZERO)
685    }
686}

727/// An anchor in time which can be used to create new `SystemTime` instances or
728/// learn about where in time a `SystemTime` lies.
729// NOTE! this documentation is duplicated, here and in SystemTime::UNIX_EPOCH.
730// The two copies are not quite identical, because of the difference in naming.
731/// This constant is defined to be "1970-01-01 00:00:00 UTC" on all systems with
732/// respect to the system clock. Using `duration_since` on an existing
733/// [`SystemTime`] instance can tell how far away from this point in time a
734/// measurement lies, and using `UNIX_EPOCH + duration` can be used to create a
735/// [`SystemTime`] instance to represent another fixed point in time.
736///
737/// `duration_since(UNIX_EPOCH).unwrap().as_secs()` returns
738/// the number of non-leap seconds since the start of 1970 UTC.
739/// This is a POSIX `time_t` (as a `u64`),
740/// and is the same time representation as used in many Internet protocols.
741///
742/// # Examples
743///
744/// ```no_run
745/// use pspsdk::time::{SystemTime, UNIX_EPOCH};
746///
747/// match SystemTime::now().duration_since(UNIX_EPOCH) {
748///     Ok(n) => pspsdk::println!("1970-01-01 00:00:00 UTC was {} seconds ago!", n.as_secs()),
749///     Err(_) => panic!("SystemTime before UNIX EPOCH!"),
750/// }
751/// ```
752pub const UNIX_EPOCH: SystemTime = SystemTime(time::UNIX_EPOCH);
753
754impl SystemTimeError {
755    /// Returns the positive duration which represents how far forward the
756    /// second system time was from the first.
757    ///
758    /// A `SystemTimeError` is returned from the [`SystemTime::duration_since`]
759    /// and [`SystemTime::elapsed`] methods whenever the second system time
760    /// represents a point later in time than the `self` of the method call.
761    ///
762    /// # Examples
763    ///
764    /// ```no_run
765    /// use pspsdk::thread::sleep;
766    ///
767    /// use pspsdk::time::{Duration, SystemTime};
768    ///
769    /// let sys_time = SystemTime::now();
770    /// sleep(Duration::from_secs(1));
771    /// let new_sys_time = SystemTime::now();
772    /// match sys_time.duration_since(new_sys_time) {
773    ///     Ok(_) => {},
774    ///     Err(e) => pspsdk::println!("SystemTimeError difference: {:?}", e.duration()),
775    /// }
776    /// ```
777    #[must_use]
778    pub fn duration(&self) -> Duration {
779        self.0
780    }
781}
