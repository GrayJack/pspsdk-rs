1//! Traits, helpers, and type definitions for core I/O functionality.
2//!
3//! The `pspsdk::io` module contains a number of common things you'll need
4//! when doing input and output. The most core part of this module is
5//! the [`Read`] and [`Write`] traits, which provide the
6//! most general interface for reading and writing input and output.
7//!
8//! The traits and most types are re-export of the [`io_core`] crate.
9//!
10//! ## Read and Write
11//!
12//! Because they are traits, [`Read`] and [`Write`] are implemented by a number
13//! of other types, and you can implement them for your types too. As such,
14//! you'll see a few different types of I/O throughout the documentation in
15//! this module: [`File`]s, [`TcpStream`]s, and sometimes even [`Vec<T>`]s. For
16//! example, [`Read`] adds a [`read`][`Read::read`] method, which we can use on
17//! [`File`]s:
18//!
19//! ```no_run
20//! use pspsdk::{
21//!     fs::File,
22//!     io::{self, Read},
23//!     println,
24//! };
25//!
26//! fn main() -> io::Result<()> {
27//!     let mut f = File::open("foo.txt")?;
28//!     let mut buffer = [0; 10];
29//!
30//!     // read up to 10 bytes
31//!     let n = f.read(&mut buffer)?;
32//!
33//!     println!("The bytes: {:?}", &buffer[..n]);
34//!     Ok(())
35//! }
36//! ```
37//!
38//! [`Read`] and [`Write`] are so important, implementors of the two traits have a
39//! nickname: readers and writers. So you'll sometimes see 'a reader' instead
40//! of 'a type that implements the [`Read`] trait'. Much easier!
41//!
42//! ## Seek and BufRead
43//!
44//! Beyond that, there are two important traits that are provided: [`Seek`]
45//! and [`BufRead`]. Both of these build on top of a reader to control
46//! how the reading happens. [`Seek`] lets you control where the next byte is
47//! coming from:
48//!
49//! ```no_run
50//! use pspsdk::{
51//!     fs::File,
52//!     io::{self, Read, SeekFrom},
53//!     println,
54//! };
55//!
56//! fn main() -> io::Result<()> {
57//!     let mut f = File::open("foo.txt")?;
58//!     let mut buffer = [0; 10];
59//!
60//!     // skip to the last 10 bytes of the file
61//!     f.seek(SeekFrom::End(-10))?;
62//!
63//!     // read up to 10 bytes
64//!     let n = f.read(&mut buffer)?;
65//!
66//!     println!("The bytes: {:?}", &buffer[..n]);
67//!     Ok(())
68//! }
69//! ```
70//!
71//! [`BufRead`] uses an internal buffer to provide a number of other ways to read, but
72//! to show it off, we'll need to talk about buffers in general. Keep reading!
73//!
74//! ## BufReader and BufWriter
75//!
76//! Byte-based interfaces are unwieldy and can be inefficient, as we'd need to be
77//! making near-constant calls to the operating system. To help with this,
78//! `pspsdk::io` comes with two structs, [`BufReader`] and [`BufWriter`], which wrap
79//! readers and writers. The wrapper uses a buffer, reducing the number of
80//! calls and providing nicer methods for accessing exactly what you want.
81//!
82//! For example, [`BufReader`] works with the [`BufRead`] trait to add extra
83//! methods to any reader:
84//!
85//! ```no_run
86//! use pspsdk::{
87//!     fs::File,
88//!     io,
89//!     io::{BufRead, BufReader, Read},
90//!     println,
91//! };
92//!
93//! fn main() -> io::Result<()> {
94//!     let f = File::open("foo.txt")?;
95//!     let mut reader = BufReader::new(f);
96//!     let mut buffer = String::new();
97//!
98//!     // read a line into buffer
99//!     reader.read_line(&mut buffer)?;
100//!
101//!     println!("{buffer}");
102//!     Ok(())
103//! }
104//! ```
105//!
106//! [`BufWriter`] doesn't add any new ways of writing; it just buffers every call
107//! to [`write`][`Write::write`]:
108//!
109//! ```no_run
110//! use pspsdk::{
111//!     fs::File,
112//!     io,
113//!     io::{BufWriter, Write},
114//!     println,
115//! };
116//!
117//! fn main() -> io::Result<()> {
118//!     let f = File::create("foo.txt")?;
119//!     {
120//!         let mut writer = BufWriter::new(f);
121//!
122//!         // write a byte to the buffer
123//!         writer.write(&[42])?;
124//!     } // the buffer is flushed once writer goes out of scope
125//!
126//!     Ok(())
127//! }
128//! ```
129//!
130//! ## Standard input and output
131//!
132//! A very common source of input is standard input:
133//!
134//! ```no_run
135//! use pspsdk::io;
136//!
137//! fn main() -> io::Result<()> {
138//!     let mut input = String::new();
139//!
140//!     io::stdin().read_line(&mut input)?;
141//!
142//!     println!("You typed: {}", input.trim());
143//!     Ok(())
144//! }
145//! ```
146//!
147//! Note that you cannot use the [`?` operator] in functions that do not return
148//! a [`Result<T, E>`][`Result`]. Instead, you can call [`.unwrap()`]
149//! or `match` on the return value to catch any possible errors:
150//!
151//! ```no_run
152//! use pspsdk::io;
153//!
154//! let mut input = String::new();
155//!
156//! io::stdin().read_line(&mut input).unwrap();
157//! ```
158//!
159//! And a very common source of output is standard output:
160//!
161//! ```no_run
162//! use pspsdk::{io, Write};
163//!
164//! fn main() -> io::Result<()> {
165//!     io::stdout().write(&[42])?;
166//!     Ok(())
167//! }
168//! ```
169//!
170//! Of course, using [`io::stdout`] directly is less common than something like
171//! [`println!`](crate::println).
172//!
173//! ## Iterator types
174//!
175//! A large number of the structures provided by `pspsdk::io` are for various
176//! ways of iterating over I/O. For example, [`Lines`] is used to split over
177//! lines:
178//!
179//! ```no_run
180//! use pspsdk::{
181//!     fs::File,
182//!     io,
183//!     io::{BufRead, BufReader, Read},
184//!     println,
185//! };
186//!
187//! fn main() -> io::Result<()> {
188//!     let f = File::open("foo.txt")?;
189//!     let reader = BufReader::new(f);
190//!
191//!     for line in reader.lines() {
192//!         println!("{}", line?);
193//!     }
194//!     Ok(())
195//! }
196//! ```
197//!
198//! ## Functions
199//!
200//! There are a number of [functions][functions-list] that offer access to various
201//! features. For example, we can use three of these functions to copy everything
202//! from standard input to standard output:
203//!
204//! ```no_run
205//! use pspsdk::io;
206//!
207//! fn main() -> io::Result<()> {
208//!     io::copy(&mut io::stdin(), &mut io::stdout())?;
209//!     Ok(())
210//! }
211//! ```
212//!
213//! [functions-list]: #functions-1
214//!
215//! ## io::Result
216//!
217//! Last, but certainly not least, is [`io::Result`]. This type is used
218//! as the return type of many `pspsdk::io` functions that can cause an error, and
219//! can be returned from your own functions as well. Many of the examples in this
220//! module use the [`?` operator]:
221//!
222//! ```
223//! use pspsdk::{io, println};
224//!
225//! fn read_input() -> io::Result<()> {
226//!     let mut input = String::new();
227//!
228//!     io::stdin().read_line(&mut input)?;
229//!
230//!     println!("You typed: {}", input.trim());
231//!
232//!     Ok(())
233//! }
234//! ```
235//!
236//! The return type of `read_input()`, [`io::Result<()>`][`io::Result`], is a very
237//! common type for functions which don't have a 'real' return value, but do want to
238//! return errors if they happen. In this case, the only purpose of this function is
239//! to read the line and print it, so we use `()`.
240//!
241//! ## I/O Safety
242//!
243//! Rust follows an I/O safety discipline that is comparable to its memory safety discipline. This
244//! means that file descriptors can be *exclusively owned*. (Here, "file descriptor" is meant to
245//! subsume similar concepts that exist across a wide range of operating systems even if they might
246//! use a different name, such as "handle".) An exclusively owned file descriptor is one that no
247//! other code is allowed to access in any way, but the owner is allowed to access and even close
248//! it any time. A type that owns its file descriptor should usually close it in its `drop`
249//! function. Types like [`File`] own their file descriptor. Similarly, file descriptors
250//! can be *borrowed*, granting the temporary right to perform operations on this file descriptor.
251//! This indicates that the file descriptor will not be closed for the lifetime of the borrow, but
252//! it does *not* imply any right to close this file descriptor, since it will likely be owned by
253//! someone else.
254//!
255//! As this try to be as close to the Rust standard library as possible, the PSP platform-specific
256//! parts of the library lives at [`os`](crate::os) module.
257//!
258//! To uphold I/O safety, it is crucial that no code acts on file descriptors it does not own or
259//! borrow, and no code closes file descriptors it does not own. In other words, a safe function
260//! that takes a regular integer, treats it as a file descriptor, and acts on it, is *unsound*.
261//!
262//! Not upholding I/O safety and acting on a file descriptor without proof of ownership can lead to
263//! misbehavior and even Undefined Behavior in code that relies on ownership of its file
264//! descriptors: a closed file descriptor could be re-allocated, so the original owner of that file
265//! descriptor is now working on the wrong file. Some code might even rely on fully encapsulating
266//! its file descriptors with no operations being performed by any other part of the program.
267//!
268//! Note that exclusive ownership of a file descriptor does *not* imply exclusive ownership of the
269//! underlying kernel object that the file descriptor references (also called "open file
270//! description" on some operating systems). File descriptors basically work like [`Arc`]: when you
271//! receive an owned file descriptor, you cannot know whether there are any other file descriptors
272//! that reference the same kernel object. However, when you create a new kernel object, you know
273//! that you are holding the only reference to it. Just be careful not to lend it to anyone, since
274//! they can obtain a clone and then you can no longer know what the reference count is! In that
275//! sense, [`OwnedFd`] is like `Arc` and [`BorrowedFd<'a>`] is like `&'a Arc` (and similar for the
276//! Windows types). In particular, given a `BorrowedFd<'a>`, you are not allowed to close the file
277//! descriptor -- just like how, given a `&'a Arc`, you are not allowed to decrement the reference
278//! count and potentially free the underlying object. There is no equivalent to `Box` for file
279//! descriptors in the standard library (that would be a type that guarantees that the reference
280//! count is `1`), however, it would be possible for a crate to define a type with those semantics.
281//!
282//! [`File`]: crate::fs::File
283//! [`TcpStream`]: crate::net::TcpStream
284//! [`io::stdout`]: stdout
285//! [`io::Result`]: self::Result
286//! [`?` operator]: https://doc.rust-lang.org/book/appendix-02-operators.html
287//! [`Result`]: core::result::Result
288//! [`.unwrap()`]: core::result::Result::unwrap

333                ErrorKind::AddrInUse => Self::ADDR_IN_USE,
334                ErrorKind::AddrNotAvailable => Self::ADDR_NOT_AVAILABLE,
335                ErrorKind::NetworkDown => Self::NETWORK_DOWN,
336                ErrorKind::BrokenPipe => Self::BROKEN_PIPE,
337                ErrorKind::AlreadyExists => Self::FILE_ALREADY_EXISTS,
338                ErrorKind::WouldBlock => Self::STD_ALREADY,

353                ErrorKind::ResourceBusy => Self::DEVICE_BUSY,

382pub(crate) fn decode_error_kind(error: i32) -> ErrorKind {
383    use ErrorKind;
384
385    match error as u32 {
386        0x00..=0x80000000 => ErrorKind::Uncategorized,
387        // Safety: It is in the valid range
388        _ => match unsafe { SceError::from_raw_unchecked(error as u32) } {
389            SceError::FILE_NOT_FOUND => ErrorKind::NotFound,
390            SceError::NO_PERM => ErrorKind::PermissionDenied,
391            SceError::CONNECTION_REFUSED => ErrorKind::ConnectionRefused,
392            SceError::CONNECTION_RESET => ErrorKind::ConnectionReset,
393            SceError::HOST_UNREACHABLE => ErrorKind::HostUnreachable,
394            SceError::NETWORK_UNREACHABLE => ErrorKind::NetworkUnreachable,
395            SceError::CONNECTION_ABORTED => ErrorKind::ConnectionAborted,
396            SceError::NOT_CONNECTED => ErrorKind::NotConnected,
397            SceError::ADDR_IN_USE => ErrorKind::AddrInUse,
398            SceError::ADDR_NOT_AVAILABLE => ErrorKind::AddrNotAvailable,
399            SceError::NETWORK_DOWN => ErrorKind::NetworkDown,
400            SceError::BROKEN_PIPE => ErrorKind::BrokenPipe,
401            SceError::FILE_ALREADY_EXISTS => ErrorKind::AlreadyExists,
402            SceError::STD_ALREADY => ErrorKind::WouldBlock,
403            SceError::NOT_A_DIRECTORY => ErrorKind::NotADirectory,
404            SceError::IS_DIRECTORY => ErrorKind::IsADirectory,
405            SceError::DIRECTORY_IS_NOT_EMPTY => ErrorKind::DirectoryNotEmpty,
406            SceError::READ_ONLY => ErrorKind::ReadOnlyFilesystem,
407            SceError::STALE_NETWORK_HANDLE => ErrorKind::StaleNetworkFileHandle,
408            SceError::STD_INVALID_ARGUMENT => ErrorKind::InvalidInput,
409            SceError::TIMEOUT => ErrorKind::TimedOut,
410            SceError::NO_FREE_BUF_SPACE => ErrorKind::WriteZero,
411            SceError::NO_SPACE => ErrorKind::StorageFull,
412            SceError::NOT_SEEKABLE => ErrorKind::NotSeekable,
413            SceError::FILE_QUOTA_EXCEEDED => ErrorKind::QuotaExceeded,
414            SceError::FILE_IS_TOO_BIG => ErrorKind::FileTooLarge,
415            SceError::DEVICE_BUSY => ErrorKind::ResourceBusy,
416            SceError::BUSY => ErrorKind::ExecutableFileBusy,
417            SceError::DEADLOCK => ErrorKind::Deadlock,
418            SceError::CROSS_DEV_LINK => ErrorKind::CrossesDevices,
419            SceError::TOO_MANY_SYMBOLIC_LINKS => ErrorKind::TooManyLinks,
420            SceError::INVALID_NAME => ErrorKind::InvalidFilename,
421            SceError::ARG_LIST_TOO_LONG => ErrorKind::ArgumentListTooLong,
422            SceError::INTERRUPTED => ErrorKind::Interrupted,
423            SceError::NOT_SUPPORTED => ErrorKind::Unsupported,
424            SceError::NO_MEMORY => ErrorKind::OutOfMemory,
425            SceError::IN_PROGRESS => ErrorKind::InProgress,
426            x if x >= SceError::OPERATION_NOT_PERMITTED && x <= SceError::WRONG_MEDIUM => {
427                ErrorKind::Other
428            },
429            _ => ErrorKind::Uncategorized,
430        },
431    }
432}

434pub(crate) fn is_interrupted(error: i32) -> bool {
435    error as u32 == SceError::INTERRUPTED.to_inner()
436}

438/// Expand for other error facilities
439pub(crate) fn error_str(error: i32) -> &'static str {
440    match error as u32 {
441        0x00..=0x80000000 => "",
442        // Safety: It is in the valid range
443        err => unsafe { SceError::from_raw_unchecked(err).error_msg() },
444    }
445}
446
447pub(crate) fn error_string(error: i32) -> String {
448    error_str(error).to_string()
449}

460impl<T: SceIntoOkValue, E: Into<SceError>> From<core::result::Result<T, E>> for SceResult<T> {
461    fn from(value: core::result::Result<T, E>) -> Self {
462        match value {
463            Ok(v) => SceResult::new(v.into_ok_value()),
464            Err(e) => SceResult::new(e.into().to_inner()),
465        }
